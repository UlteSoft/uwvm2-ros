cmake_minimum_required(VERSION 3.16)
project(uwvm_multiarch_native LANGUAGES CXX)
if(NOT IS_ABSOLUTE "${UWVM_MULTIARCH_SOURCE}" OR NOT EXISTS "${UWVM_MULTIARCH_SOURCE}/gateways")
  message(FATAL_ERROR "Select the exact complete frozen UWVM_MULTIARCH_SOURCE")
endif()
if(NOT UWVM_MULTIARCH_BACKEND MATCHES "^[12]$")
  message(FATAL_ERROR "Select exactly one native ROS backend")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(UWVM_ROS2_BUILD_PROBES OFF CACHE BOOL "Only explicit native peers in this separate lane" FORCE)
add_subdirectory("${UWVM_MULTIARCH_SOURCE}/gateways/ros${UWVM_MULTIARCH_BACKEND}" gateway)
set(gateway "uwvm-ros${UWVM_MULTIARCH_BACKEND}-gateway")
target_compile_options(${gateway} PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/stop.h" -UNDEBUG)
foreach(role IN ITEMS sdk events)
  if(role STREQUAL sdk)
    set(native_source "${UWVM_MULTIARCH_SOURCE}/tests/ros_sdk_relay/native_ros${UWVM_MULTIARCH_BACKEND}_peer.cc")
  else()
    set(native_source "${UWVM_MULTIARCH_SOURCE}/tests/ros_unified_events/native_events_ros${UWVM_MULTIARCH_BACKEND}.cc")
  endif()
  set(target "uwvm-multiarch-${role}-peer")
  add_executable(${target} "${native_source}")
  target_compile_options(${target} PRIVATE -UNDEBUG)
  target_include_directories(${target} PRIVATE "${UWVM_MULTIARCH_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/gateway")
  if(UWVM_MULTIARCH_BACKEND STREQUAL "1")
    # Consume the Gateway's actual compile/link-admitted SDK in either CATKIN
    # or EXPLICIT mode. Its interface also carries the selected string ABI and
    # any official standard-service header overlay; a second catkin lookup
    # would discard that binding and cannot relocate host-absolute SDK exports.
    if(NOT TARGET uwvm-ros1-sdk)
      message(FATAL_ERROR "The actual ROS1 Gateway SDK interface is required")
    endif()
    target_link_libraries(${target} PRIVATE uwvm-ros1-sdk)
  else()
    find_package(rclcpp REQUIRED)
    find_package(rclcpp_action REQUIRED)
    find_package(action_tutorials_interfaces REQUIRED)
    find_package(std_msgs REQUIRED)
    find_package(std_srvs REQUIRED)
    uwvm_ros2_dependencies(${target} rclcpp rclcpp_action action_tutorials_interfaces std_msgs std_srvs)
    target_link_libraries(${target} pthread)
  endif()
endforeach()
