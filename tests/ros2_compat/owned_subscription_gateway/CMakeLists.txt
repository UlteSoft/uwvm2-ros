cmake_minimum_required(VERSION 3.20)
project(uwvm_owned_subscription_gateway_contracts LANGUAGES CXX)
set(UWVM_SOURCE_ROOT "" CACHE PATH "Exact composed reviewed source root")
set(UWVM_ROS2_GATEWAY_FEATURE_DIR "" CACHE PATH "Actual SDK compile/link feature directory")
option(UWVM_OWNED_SUBSCRIPTION_WIRE_MODELS "Build actual socket/synthetic wire cleanup contracts" OFF)
option(UWVM_OWNED_SUBSCRIPTION_NATIVE_WITNESS "Build independent original-SDK native client" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "These actual peer-credential/pidfd/Socket contracts require Linux")
endif()
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
  message(FATAL_ERROR "Choose an actual admitted C++ compiler standard")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/gateways/common/server.h" OR
   NOT EXISTS "${UWVM_SOURCE_ROOT}/src/uwvm2/imported/ros/ipc/rmw_subscription.h" OR
   NOT EXISTS "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include/fast_io_freestanding.h")
  message(FATAL_ERROR "Exact integrated server/UMRS and actual vendored fast_io headers are required")
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
function(source_contract target)
  target_include_directories(${target} PRIVATE "${UWVM_SOURCE_ROOT}/src"
    "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include")
  target_link_libraries(${target} Threads::Threads OpenSSL::Crypto)
endfunction()
if(UWVM_OWNED_SUBSCRIPTION_WIRE_MODELS)
  add_executable(owned-subscription-gateway-wire server_contract.cc)
  source_contract(owned-subscription-gateway-wire)
endif()
if(UWVM_OWNED_SUBSCRIPTION_NATIVE_WITNESS)
  if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
    message(FATAL_ERROR "Actual original-SDK compile/link features required; never force native feature flags")
  endif()
  find_package(ament_cmake REQUIRED)
  foreach(dependency IN ITEMS rclcpp rcl rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    find_package(${dependency} REQUIRED)
  endforeach()
  add_executable(owned-subscription-gateway-native native_gateway_contract.cc)
  source_contract(owned-subscription-gateway-native)
  target_include_directories(owned-subscription-gateway-native PRIVATE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
  ament_target_dependencies(owned-subscription-gateway-native rclcpp rcl rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
endif()
# No configure-time executable, add_test or auto-Gateway launch. A separately
# admitted original SDK/MD/link/native process owner must execute any command.
