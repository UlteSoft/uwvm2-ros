cmake_minimum_required(VERSION 3.16)
project(uwvm_industrial_gateway_witness LANGUAGES CXX)
option(UWVM_BUILD_INDUSTRIAL_GATEWAY_WITNESS "Build explicit original native industrial peers" OFF)
if(NOT UWVM_BUILD_INDUSTRIAL_GATEWAY_WITNESS)
  message(FATAL_ERROR "Explicit UWVM_BUILD_INDUSTRIAL_GATEWAY_WITNESS=ON is required")
endif()
foreach(path UWVM_INDUSTRIAL_WITNESS_VALUES UWVM_FAST_IO_INCLUDE UWVM_ROS_CUSTOM_REGISTRY_HEADER)
  if(NOT IS_ABSOLUTE "${${path}}" OR NOT EXISTS "${${path}}" OR "${${path}}" MATCHES "[;\"\\\r\n]")
    message(FATAL_ERROR "${path} must name an explicit safe physical input")
  endif()
endforeach()
if(NOT UWVM_NATIVE_PACKAGES)
  message(FATAL_ERROR "Explicit original native package selection required")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
foreach(package IN LISTS UWVM_NATIVE_PACKAGES)
  if(NOT package MATCHES "^[a-z][a-z0-9_]*$")
    message(FATAL_ERROR "Unsafe native package name")
  endif()
  find_package(${package} REQUIRED)
endforeach()
file(GLOB cases CONFIGURE_DEPENDS "${UWVM_INDUSTRIAL_WITNESS_VALUES}/case_*.cc")
if(NOT cases OR NOT EXISTS "${UWVM_INDUSTRIAL_WITNESS_VALUES}/cases.h")
  message(FATAL_ERROR "Complete original value preparation required")
endif()
add_executable(uwvm-industrial-gateway-native native_session.cc ${cases})
target_compile_features(uwvm-industrial-gateway-native PRIVATE cxx_std_20)
target_compile_options(uwvm-industrial-gateway-native PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
target_include_directories(uwvm-industrial-gateway-native PRIVATE . ../../.. ../../../src ../../../sdk/include
  "${UWVM_INDUSTRIAL_WITNESS_VALUES}" "${UWVM_FAST_IO_INCLUDE}")
target_compile_definitions(uwvm-industrial-gateway-native PRIVATE
  UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_ROS_CUSTOM_REGISTRY_HEADER}")
ament_target_dependencies(uwvm-industrial-gateway-native rclcpp rclcpp_action rcl_interfaces ${UWVM_NATIVE_PACKAGES})
target_link_libraries(uwvm-industrial-gateway-native pthread)
