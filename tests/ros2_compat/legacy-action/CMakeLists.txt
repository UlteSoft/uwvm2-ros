cmake_minimum_required(VERSION 3.16)
project(uwvm_legacy_action_native_contract LANGUAGES CXX)
# Configuration and every process belong to the real original-pidfd owner.
set(UWVM_SOURCE "" CACHE PATH "Actual complete frozen current source")
set(UWVM_LEGACY_ACTION_FIXTURE_PACKAGE "" CACHE STRING "Exactly example_interfaces or action_tutorials_interfaces")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/gateways/ros2/compat/metadata_export.h")
  message(FATAL_ERROR "Actual complete current source required")
endif()
find_package(ament_cmake REQUIRED)
find_package(action_msgs REQUIRED)
find_package(unique_identifier_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(UWVM_LEGACY_ACTION_FIXTURE_PACKAGE STREQUAL "example_interfaces")
  find_package(example_interfaces REQUIRED)
  set(fixture_flag 1)
elseif(UWVM_LEGACY_ACTION_FIXTURE_PACKAGE STREQUAL "action_tutorials_interfaces")
  find_package(action_tutorials_interfaces REQUIRED)
  set(fixture_flag 0)
else()
  message(FATAL_ERROR "Select an actually installed official Fibonacci package; no guessed distro fallback")
endif()
add_executable(legacy-action-native-contract native_contract.cc)
target_compile_features(legacy-action-native-contract PRIVATE cxx_std_17)
target_compile_definitions(legacy-action-native-contract PRIVATE UWVM_ROS2_HAVE_OPENSSL=1 UWVM_LEGACY_ACTION_EXAMPLE_INTERFACES=${fixture_flag})
target_include_directories(legacy-action-native-contract PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/src")
ament_target_dependencies(legacy-action-native-contract ${UWVM_LEGACY_ACTION_FIXTURE_PACKAGE} action_msgs unique_identifier_msgs builtin_interfaces rosidl_typesupport_cpp rosidl_typesupport_introspection_cpp)
target_link_libraries(legacy-action-native-contract PRIVATE OpenSSL::Crypto)
target_compile_options(legacy-action-native-contract PRIVATE -O1 -g0 -UNDEBUG -std=gnu++17 -MD
  -MF "${CMAKE_BINARY_DIR}/CMakeFiles/legacy-action-native-contract.dir/native_contract.cc.o.d"
  -MT "${CMAKE_BINARY_DIR}/CMakeFiles/legacy-action-native-contract.dir/native_contract.cc.o")
target_link_options(legacy-action-native-contract PRIVATE -Wl,-t)
