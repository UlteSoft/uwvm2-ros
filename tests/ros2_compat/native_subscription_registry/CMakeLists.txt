cmake_minimum_required(VERSION 3.20)
project(uwvm_subscription_registry_contract LANGUAGES CXX)
option(UWVM_SUBSCRIPTION_REGISTRY_WIRE "Build SDK-free value contract" OFF)
option(UWVM_SUBSCRIPTION_REGISTRY_REAL_SDK "Build actual original-SDK witness" OFF)
get_filename_component(UWVM_SUBSCRIPTION_REGISTRY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
  message(FATAL_ERROR "Select a real admitted C++ standard explicitly")
endif()
find_package(Threads REQUIRED)
function(uwvm_subscription_checks target)
  target_include_directories(${target} PRIVATE "${UWVM_SUBSCRIPTION_REGISTRY_ROOT}/src")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
endfunction()
if(UWVM_SUBSCRIPTION_REGISTRY_WIRE)
  add_executable(subscription-registry-wire wire_contract.cc)
  uwvm_subscription_checks(subscription-registry-wire)
endif()
if(UWVM_SUBSCRIPTION_REGISTRY_REAL_SDK)
  if(NOT CMAKE_CXX_STANDARD STREQUAL "17")
    message(FATAL_ERROR "This original-Gateway SDK witness requires its actual C++17 native profile")
  endif()
  find_package(ament_cmake REQUIRED)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    find_package(${dependency} REQUIRED)
  endforeach()
  if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
    message(FATAL_ERROR "Supply the actual selected-SDK compile/link-probed D4 feature header")
  endif()
  add_executable(subscription-registry-native native_registry_contract.cc)
  uwvm_subscription_checks(subscription-registry-native)
  target_include_directories(subscription-registry-native PRIVATE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
  target_link_libraries(subscription-registry-native PRIVATE OpenSSL::Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    ament_target_dependencies(subscription-registry-native ${dependency})
  endforeach()
endif()

# admit each actual tool/SDK, namespace/ROS endpoint and complete teardown.
