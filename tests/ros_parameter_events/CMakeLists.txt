cmake_minimum_required(VERSION 3.16)
project(uwvm_expanded_parameter_events LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
foreach(target native_ros2 witness_ros2)
  add_executable(${target} ${target}.cc)
  target_compile_options(${target} PRIVATE -UNDEBUG)
  target_include_directories(${target} PRIVATE ${UWVM_SOURCE}/src ${UWVM_SOURCE}/third-parties/fast_io/include)
  ament_target_dependencies(${target} rclcpp rcl_interfaces)
endforeach()
