cmake_minimum_required(VERSION 3.16)
project(uwvm_ros_local_parameter_events LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rcl_action REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
foreach(target native_ros2 contract_ros2)
  add_executable(local_parameter_events_${target} ${target}.cc)
  target_compile_features(local_parameter_events_${target} PRIVATE cxx_std_20)
  target_compile_options(local_parameter_events_${target} PRIVATE -UNDEBUG)
  target_include_directories(local_parameter_events_${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../src ${CMAKE_CURRENT_SOURCE_DIR}/../../third-parties/fast_io/include)
  ament_target_dependencies(local_parameter_events_${target} rclcpp rcl_interfaces rcl_action rosidl_typesupport_cpp)
  target_link_libraries(local_parameter_events_${target} pthread)
endforeach()
