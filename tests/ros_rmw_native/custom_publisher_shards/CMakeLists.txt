cmake_minimum_required(VERSION 3.20)
project(uwvm_custom_owned_publisher_contract LANGUAGES CXX)
# This is a separately owned original-SDK compile/link contract. No CTest or
# configure probe launches an actual native/ROS/Guest process.
set(UWVM_SOURCE_ROOT "" CACHE PATH "Exact integrated source root")
set(UWVM_ROS2_GATEWAY_FEATURE_DIR "" CACHE PATH "Actual selected SDK compile/link feature directory")
set(UWVM_ROS_CUSTOM_REGISTRY_HEADER "" CACHE FILEPATH "Exact complete generated manifest")
set(UWVM_ROS_CUSTOM_ROS2_HEADER "" CACHE FILEPATH "Exact complete generated ROS2 native provider")
set(UWVM_ROS_CUSTOM_ROS2_PACKAGES "" CACHE STRING "Explicit installed original interface packages")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/gateways/ros2/custom_topics.cmake")
 message(FATAL_ERROR "Supply the actual integrated source root")
endif()
# parameter.h selects the real fast_io freestanding dependency when concepts
# are enabled. Resolve it from this exact source tree for every admitted C++
# standard; an ambient include path must not hide an incomplete build recipe.
set(uwvm_fast_io_include "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include")
if(NOT EXISTS "${uwvm_fast_io_include}/fast_io_freestanding.h")
 message(FATAL_ERROR "The exact integrated source root must contain vendored fast_io headers")
endif()
if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
 message(FATAL_ERROR "Supply actual selected-SDK Gateway compile/link features")
endif()
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
 message(FATAL_ERROR "Select the actual admitted compiler language standard explicitly")
endif()
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
foreach(package IN ITEMS rclcpp rcl rmw rosidl_typesupport_introspection_cpp)
 find_package(${package} REQUIRED)
endforeach()
add_executable(custom-owned-publisher-contract ../custom_publisher_factory_contract.cc)
target_include_directories(custom-owned-publisher-contract PRIVATE "${UWVM_SOURCE_ROOT}/src"
 "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" "${uwvm_fast_io_include}")
# Use the original plain ament linkage convention consistently (old SDKs too).
target_link_libraries(custom-owned-publisher-contract Threads::Threads OpenSSL::Crypto)
ament_target_dependencies(custom-owned-publisher-contract rclcpp rcl rmw rosidl_typesupport_introspection_cpp)
if(UWVM_ROS_CUSTOM_REGISTRY_HEADER OR UWVM_ROS_CUSTOM_ROS2_HEADER OR UWVM_ROS_CUSTOM_ROS2_PACKAGES)
 if(NOT UWVM_ROS_CUSTOM_REGISTRY_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_PACKAGES)
  message(FATAL_ERROR "A custom build requires the whole paired registry/provider and explicit package list")
 endif()
 foreach(header IN ITEMS UWVM_ROS_CUSTOM_REGISTRY_HEADER UWVM_ROS_CUSTOM_ROS2_HEADER)
  if(NOT IS_ABSOLUTE "${${header}}" OR NOT EXISTS "${${header}}" OR IS_DIRECTORY "${${header}}")
   message(FATAL_ERROR "${header} must select an existing absolute actual generated header")
  endif()
    if("${${header}}" MATCHES "[;\"\\\\\r\n]")
   message(FATAL_ERROR "Unsupported quoted include path")
  endif()
 endforeach()
 set(seen "")
 foreach(package IN LISTS UWVM_ROS_CUSTOM_ROS2_PACKAGES)
  if(NOT package MATCHES "^[a-z][a-z0-9_]*$")
   message(FATAL_ERROR "Select actual interface package identifiers")
  endif()
  list(FIND seen "${package}" duplicate)
  if(NOT duplicate EQUAL -1)
   message(FATAL_ERROR "Duplicate selected custom package")
  endif()
  list(APPEND seen "${package}")
  find_package(${package} REQUIRED)
 endforeach()
 target_compile_definitions(custom-owned-publisher-contract PRIVATE
  UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_ROS_CUSTOM_REGISTRY_HEADER}"
  UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_ROS_CUSTOM_ROS2_HEADER}")
 ament_target_dependencies(custom-owned-publisher-contract ${UWVM_ROS_CUSTOM_ROS2_PACKAGES})
 include("${UWVM_SOURCE_ROOT}/gateways/ros2/custom_topics.cmake")
 uwvm_ros2_custom_topic_sources(custom-owned-publisher-contract)
endif()
# Empty provider deliberately links no dispatch/shard objects. The public
# if-constexpr empty route must not need an unavailable external factory symbol.
