cmake_minimum_required(VERSION 3.16)
project(uwvm_native_rmw_contracts LANGUAGES CXX)
# Targets are explicit; no CTest invocation starts unowned child processes.
option(UWVM_NATIVE_RMW_MODELS "Build wire and existing Image policy contracts" OFF)
option(UWVM_NATIVE_RMW_REAL_SDK "Build the original native publisher witness" OFF)
get_filename_component(UWVM_NATIVE_RMW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
  message(FATAL_ERROR "Select the admitted actual compiler standard explicitly")
endif()
if(UWVM_NATIVE_RMW_MODELS)
  foreach(name IN ITEMS wire image_policy)
    add_executable(native-rmw-${name}-contract ${name}_contract.cc)
    target_include_directories(native-rmw-${name}-contract PRIVATE "${UWVM_NATIVE_RMW_ROOT}/src")
    if(MSVC)
      target_compile_options(native-rmw-${name}-contract PRIVATE /UNDEBUG)
    else()
      target_compile_options(native-rmw-${name}-contract PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
if(UWVM_NATIVE_RMW_REAL_SDK)
  find_package(ament_cmake REQUIRED)
  find_package(Threads REQUIRED)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    find_package(${dependency} REQUIRED)
  endforeach()
  if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
    message(FATAL_ERROR "Supply the real pinned selected-SDK Gateway feature directory")
  endif()
  add_executable(native-rmw-publisher-contract native_publisher_contract.cc)
  target_include_directories(native-rmw-publisher-contract PRIVATE "${UWVM_NATIVE_RMW_ROOT}/src" "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
  target_link_libraries(native-rmw-publisher-contract PRIVATE Threads::Threads OpenSSL::Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    if(COMMAND ament_target_dependencies)
      ament_target_dependencies(native-rmw-publisher-contract ${dependency})
    elseif(DEFINED ${dependency}_TARGETS AND NOT "${${dependency}_TARGETS}" STREQUAL "")
      target_link_libraries(native-rmw-publisher-contract PRIVATE ${${dependency}_TARGETS})
    elseif(TARGET ${dependency}::${dependency})
      target_link_libraries(native-rmw-publisher-contract PRIVATE ${dependency}::${dependency})
    else()
      message(FATAL_ERROR "Missing genuine native dependency link interface: ${dependency}")
    endif()
  endforeach()
endif()
