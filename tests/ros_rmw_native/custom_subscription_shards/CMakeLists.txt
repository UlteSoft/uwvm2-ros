cmake_minimum_required(VERSION 3.20)
project(uwvm_custom_owned_subscription_contracts LANGUAGES CXX)
# Instructions for a separately bounded real SDK owner. No configure probes,
# add_test, auto-execution or native substitute definitions are installed here.
set(UWVM_SOURCE_ROOT "" CACHE PATH "Exact composed reviewed source tree")
set(UWVM_ROS2_GATEWAY_FEATURE_DIR "" CACHE PATH "Actual SDK compile/link feature output")
set(UWVM_ROS_CUSTOM_REGISTRY_HEADER "" CACHE FILEPATH "Actual complete generated common manifest")
set(UWVM_ROS_CUSTOM_ROS2_HEADER "" CACHE FILEPATH "Actual complete generated native provider")
set(UWVM_ROS_CUSTOM_ROS2_PACKAGES "" CACHE STRING "Complete original interface package list")
option(UWVM_CUSTOM_SUBSCRIPTION_NATIVE_WITNESS "Build the explicit selected-row actual SDK witness" OFF)
set(UWVM_CUSTOM_SUBSCRIPTION_TEST_ROW "" CACHE STRING "Actual full-provider row ordinal for native witness")
set(UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD "" CACHE STRING "NEGATIVE LINK ONLY: exact one ordinal to omit")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/gateways/ros2/custom_topics.cmake")
  message(FATAL_ERROR "Supply the actual integrated source root")
endif()
if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
  message(FATAL_ERROR "Supply actual selected-SDK feature/probe results")
endif()
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
  message(FATAL_ERROR "Choose the actual admitted compiler language standard")
endif()
# parameter.h uses the genuine vendored fast_io_freestanding.h under
# __cpp_concepts. Do not depend on ambient CPLUS_INCLUDE_PATH for20/23/26.
if(NOT EXISTS "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include/fast_io_freestanding.h")
  message(FATAL_ERROR "Require the actual reviewed vendored fast_io include tree")
endif()
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(original_dependencies rclcpp rcl rmw rosidl_typesupport_introspection_cpp std_msgs geometry_msgs sensor_msgs)
foreach(package IN LISTS original_dependencies)
  find_package(${package} REQUIRED)
endforeach()
set(has_custom FALSE)
if(UWVM_ROS_CUSTOM_REGISTRY_HEADER OR UWVM_ROS_CUSTOM_ROS2_HEADER OR UWVM_ROS_CUSTOM_ROS2_PACKAGES)
  if(NOT UWVM_ROS_CUSTOM_REGISTRY_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_PACKAGES)
    message(FATAL_ERROR "Require the whole paired provider/manifest and explicit package list")
  endif()
  foreach(header IN ITEMS UWVM_ROS_CUSTOM_REGISTRY_HEADER UWVM_ROS_CUSTOM_ROS2_HEADER)
    if(NOT IS_ABSOLUTE "${${header}}" OR NOT EXISTS "${${header}}" OR IS_DIRECTORY "${${header}}")
      message(FATAL_ERROR "${header} must be an actual absolute generated header")
    endif()
    if("${${header}}" MATCHES "[;\"\\\\\r\n]")
      message(FATAL_ERROR "Unsupported quoted include path")
    endif()
  endforeach()
  set(seen "")
  foreach(package IN LISTS UWVM_ROS_CUSTOM_ROS2_PACKAGES)
    if(NOT package MATCHES "^[a-z][a-z0-9_]*$")
      message(FATAL_ERROR "Use actual original interface package identifiers")
    endif()
    list(FIND seen "${package}" duplicate)
    if(NOT duplicate EQUAL -1)
      message(FATAL_ERROR "Duplicate selected custom package")
    endif()
    list(APPEND seen "${package}")
    find_package(${package} REQUIRED)
  endforeach()
  set(has_custom TRUE)
  include("${UWVM_SOURCE_ROOT}/gateways/ros2/custom_topics.cmake")
endif()
function(prepare_contract target)
  target_include_directories(${target} PRIVATE "${UWVM_SOURCE_ROOT}/src"
    "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include" "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
  # Keep old ament plain linkage convention throughout this selected target.
  target_link_libraries(${target} Threads::Threads OpenSSL::Crypto)
  ament_target_dependencies(${target} ${original_dependencies})
  if(has_custom)
    target_compile_definitions(${target} PRIVATE
      UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_ROS_CUSTOM_REGISTRY_HEADER}"
      UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_ROS_CUSTOM_ROS2_HEADER}")
    ament_target_dependencies(${target} ${UWVM_ROS_CUSTOM_ROS2_PACKAGES})
    uwvm_ros2_custom_topic_sources(${target})
  endif()
endfunction()
add_executable(custom-owned-subscription-contract ../custom_subscription_factory_contract.cc)
prepare_contract(custom-owned-subscription-contract)
if(NOT UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD STREQUAL "")
  # This option deliberately constructs an INCOMPLETE, NONQUALIFYING negative
  # target. No normal producer may set it or treat its build failure as a pass.
  if(NOT has_custom OR NOT UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD MATCHES "^(0|[1-9][0-9]*)$" OR
     UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD GREATER_EQUAL UWVM_ROS2_CUSTOM_TOPIC_SHARDS)
    message(FATAL_ERROR "Negative omission needs a real custom build and one actual shard ordinal")
  endif()
  get_target_property(complete_sources custom-owned-subscription-contract SOURCES)
  set(omitted "${CMAKE_CURRENT_BINARY_DIR}/custom-topic-shards/custom_topics_shard_${UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD}.cc")
  list(FIND complete_sources "${omitted}" ordinal)
  if(ordinal EQUAL -1)
    message(FATAL_ERROR "Expected actual generated shard source was not present")
  endif()
  list(REMOVE_ITEM complete_sources "${omitted}")
  set_property(TARGET custom-owned-subscription-contract PROPERTY SOURCES "${complete_sources}")
  message(STATUS "NONQUALIFYING negative link target deliberately omits ${omitted}")
endif()
if(UWVM_CUSTOM_SUBSCRIPTION_NATIVE_WITNESS)
  if(NOT has_custom OR NOT UWVM_CUSTOM_SUBSCRIPTION_TEST_ROW MATCHES "^(0|[1-9][0-9]*)$" OR
     NOT UWVM_CUSTOM_SUBSCRIPTION_OMIT_SHARD STREQUAL "")
    message(FATAL_ERROR "Native witness requires complete custom sources, explicit row and no omission")
  endif()
  add_executable(custom-owned-subscription-native ../custom_subscription_native_contract.cc)
  target_compile_definitions(custom-owned-subscription-native PRIVATE UWVM_CUSTOM_SUBSCRIPTION_TEST_ROW=${UWVM_CUSTOM_SUBSCRIPTION_TEST_ROW})
  prepare_contract(custom-owned-subscription-native)
endif()
