cmake_minimum_required(VERSION 3.16)
project(uwvm_unitree_native_typed_peer LANGUAGES CXX)
# Root stages this exact source file as CMakeLists.txt in a fresh owned build
# input directory. Every package must come from the independently pinned native
# SDK and complete GO/HG/API generated installation. No download, installation,
# executable try_run, guessed distro prefix or message substitute is permitted.
set(BUILD_TESTING OFF CACHE BOOL "Only the explicit native witness" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(UWVM_SOURCE "" CACHE PATH "Complete frozen project source")
set(UWVM_UNITREE_NATIVE_CXX_STANDARD "20" CACHE STRING "Explicit SDK C++17 or20")
if(NOT UWVM_UNITREE_NATIVE_CXX_STANDARD MATCHES "^(17|20)$")
  message(FATAL_ERROR "Select actual C++17 or20 with unchanged strict float flags")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR
   NOT EXISTS "${UWVM_SOURCE}/tests/ros_unitree/native_typed_ros2_peer.cpp" OR
   NOT EXISTS "${UWVM_SOURCE}/gateways/common/native_io.h")
  message(FATAL_ERROR "An explicit complete frozen native witness source is required")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(unitree_go REQUIRED)
find_package(unitree_hg REQUIRED)
find_package(unitree_api REQUIRED)
find_package(Threads REQUIRED)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
  message(FATAL_ERROR "Actual full-MD and native link trace require GNU/Clang")
endif()
set(CMAKE_CXX_STANDARD "${UWVM_UNITREE_NATIVE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES ${rclcpp_INCLUDE_DIRS})
if(TARGET rclcpp::rclcpp)
  set(CMAKE_REQUIRED_LIBRARIES rclcpp::rclcpp)
else()
  set(CMAKE_REQUIRED_LIBRARIES ${rclcpp_LIBRARIES})
endif()
# Reprobe public quiet construction for the selected SDK; an unsupported API
# must stop configuration. This links a probe but never constructs a live node.
unset(UWVM_UNITREE_TYPED_QUIET_NODE CACHE)
check_cxx_source_compiles("#include <rclcpp/rclcpp.hpp>
#include <chrono>
#include <memory>
int main(){auto context=std::make_shared<rclcpp::Context>();
rclcpp::NodeOptions options;options.context(context).use_global_arguments(false).enable_rosout(false);
options.start_parameter_services(false).start_parameter_event_publisher(false).use_intra_process_comms(false);
options.parameter_overrides({rclcpp::Parameter(\"start_type_description_service\",false)});
auto node=std::make_shared<rclcpp::Node>(\"compile_only\",\"/uwvm_unitree\",options);
rclcpp::ExecutorOptions eo;eo.context=context;rclcpp::executors::SingleThreadedExecutor executor{eo};
executor.add_node(node);executor.spin_once(std::chrono::nanoseconds{0});executor.remove_node(node);return 0;}"
 UWVM_UNITREE_TYPED_QUIET_NODE)
if(NOT UWVM_UNITREE_TYPED_QUIET_NODE)
  message(FATAL_ERROR "Selected SDK cannot link the required quiet bounded executor")
endif()
unset(CMAKE_REQUIRED_INCLUDES)
unset(CMAKE_REQUIRED_LIBRARIES)
add_executable(unitree-native-typed-peer "${UWVM_SOURCE}/tests/ros_unitree/native_typed_ros2_peer.cpp")
target_include_directories(unitree-native-typed-peer PRIVATE "${UWVM_SOURCE}")
if(UWVM_UNITREE_NATIVE_CXX_STANDARD STREQUAL "20")
  target_include_directories(unitree-native-typed-peer PRIVATE
    "${UWVM_SOURCE}/third-parties/fast_io/include"
    "${UWVM_SOURCE}/third-parties/bizwen/include"
    "${UWVM_SOURCE}/third-parties/boost_unordered/include")
endif()
# Preserve the older ament macro's plain link signature. New ament releases
# can instead expose imported targets or explicit generated-package exports.
if(COMMAND ament_target_dependencies)
  ament_target_dependencies(unitree-native-typed-peer rclcpp unitree_go unitree_hg unitree_api)
else()
  foreach(package IN ITEMS rclcpp unitree_go unitree_hg unitree_api)
    if(TARGET ${package}::${package})
      target_link_libraries(unitree-native-typed-peer ${package}::${package})
    elseif(DEFINED ${package}_TARGETS AND NOT "${${package}_TARGETS}" STREQUAL "")
      target_link_libraries(unitree-native-typed-peer ${${package}_TARGETS})
    elseif(DEFINED ${package}_LIBRARIES AND NOT "${${package}_LIBRARIES}" STREQUAL "")
      target_include_directories(unitree-native-typed-peer SYSTEM PRIVATE ${${package}_INCLUDE_DIRS})
      target_link_libraries(unitree-native-typed-peer ${${package}_LIBRARIES})
    else()
      message(FATAL_ERROR "Selected ${package} has no supported explicit public link export")
    endif()
  endforeach()
endif()
target_link_libraries(unitree-native-typed-peer Threads::Threads)
target_compile_options(unitree-native-typed-peer PRIVATE -O3 -g0 -UNDEBUG
  -fno-fast-math -ffp-contract=off -MD
  -MF "${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/unitree-native-typed-peer.dir/native_typed_ros2_peer.cpp.o.d")
target_link_options(unitree-native-typed-peer PRIVATE -Wl,--trace)
# These are configuration facts only. Root retains actual effective compile
# argv, full dependency bodies, link traces, ELF/maps, SDK/source preservation
# and separately witnessed Runtime/native effects before claiming acceptance.
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-typed-contract.txt" CONTENT
"status=CONFIGURED_NOT_EXECUTED\ncxx_standard=${UWVM_UNITREE_NATIVE_CXX_STANDARD}\nartifact=$<TARGET_FILE:unitree-native-typed-peer>\nprotocol=UTN1/M/Q/P/A/Z/F\nfamily=GO20,HG35,API8\nrole=INDEPENDENT_NATIVE_DDS_WITNESS_NOT_WASM_OR_PHYSICS\n")
