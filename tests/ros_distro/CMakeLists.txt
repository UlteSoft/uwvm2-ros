cmake_minimum_required(VERSION 3.10)
project(uwvm_ros_distribution_basic LANGUAGES CXX)
# The caller selects the native SDK's standard and toolchain explicitly. This
# fixture does not share the interpreter's independent C++26 compiler setting.
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23)$")
  message(FATAL_ERROR "Select CMAKE_CXX_STANDARD=17,20,23 explicitly")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(UWVM_ROS_BACKEND STREQUAL "1")
  if(UWVM_ROS1_DISCOVERY STREQUAL "EXPLICIT")
    if(NOT UWVM_ROS1_INCLUDE_DIRS OR NOT UWVM_ROS1_LIBRARIES)
      message(FATAL_ERROR "EXPLICIT needs physical SDK include/library lists")
    endif()
    set(catkin_INCLUDE_DIRS ${UWVM_ROS1_INCLUDE_DIRS})
    set(catkin_LIBRARIES ${UWVM_ROS1_LIBRARIES})
  else()
    find_package(catkin REQUIRED COMPONENTS roscpp std_msgs)
  endif()
elseif(UWVM_ROS_BACKEND STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rmw REQUIRED)
  find_package(std_msgs REQUIRED)
  # The caller takes this Boolean from the pinned Gateway's actual SDK probe
  # manifest. Never guess it from the distribution or override it by a tag.
  if(NOT UWVM_BASIC_HAVE_QOS MATCHES "^[01]$")
    message(FATAL_ERROR "Pinned Gateway qos_class feature must be supplied as 0 or 1")
  endif()
else()
  message(FATAL_ERROR "UWVM_ROS_BACKEND must be exactly 1 or 2")
endif()
configure_file(basic_features.h.in basic_features.h)
add_executable(uwvm-distro-native-basic native_basic.cc)
target_include_directories(uwvm-distro-native-basic PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ../../gateways/common)
target_compile_definitions(uwvm-distro-native-basic PRIVATE UWVM_ROS_BACKEND=${UWVM_ROS_BACKEND})
target_link_libraries(uwvm-distro-native-basic Threads::Threads)
if(UWVM_ROS_BACKEND STREQUAL "1")
  target_include_directories(uwvm-distro-native-basic PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(uwvm-distro-native-basic ${catkin_LIBRARIES})
else()
  # New ament SDKs can export targets without the legacy dependency macro;
  # older SDKs expose the macro or explicit include/library variables. Require
  # an actual link interface in every case, as the Gateway compatibility entry
  # does; finding a package is not a successful empty link dependency.
  if(COMMAND ament_target_dependencies)
    ament_target_dependencies(uwvm-distro-native-basic rclcpp rmw std_msgs)
  else()
    foreach(package IN ITEMS rclcpp rmw std_msgs)
      if(DEFINED ${package}_TARGETS AND NOT "${${package}_TARGETS}" STREQUAL "")
        target_link_libraries(uwvm-distro-native-basic ${${package}_TARGETS})
      elseif(TARGET ${package}::${package})
        target_link_libraries(uwvm-distro-native-basic ${package}::${package})
      elseif(DEFINED ${package}_LIBRARIES AND NOT "${${package}_LIBRARIES}" STREQUAL "")
        target_include_directories(uwvm-distro-native-basic SYSTEM PRIVATE ${${package}_INCLUDE_DIRS})
        target_link_libraries(uwvm-distro-native-basic ${${package}_LIBRARIES})
      else()
        message(FATAL_ERROR "Selected package ${package} has no supported native link interface")
      endif()
    endforeach()
  endif()
endif()
