cmake_minimum_required(VERSION 3.16)
project(uwvm_service_availability_contracts LANGUAGES C CXX)
# No target is selected implicitly and CTest spawns no unowned children.
option(UWVM_AVAILABILITY_LOCAL_MODELS "Build actual wire/manager/server/debug model TUs, not ROS" OFF)
option(UWVM_AVAILABILITY_NATIVE "Build real selected-SDK Trigger helper witness" OFF)
get_filename_component(UWVM_AVAILABILITY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
if(UWVM_AVAILABILITY_LOCAL_MODELS)
  if(NOT CMAKE_CXX_STANDARD MATCHES "^(20|23|26)$")
    message(FATAL_ERROR "Select the actual Core model compiler standard20/23/26")
  endif()
  foreach(model IN ITEMS wire_history core server debug)
    add_executable(availability-${model}-contract ${model}_contract.cc)
    target_include_directories(availability-${model}-contract PRIVATE "${UWVM_AVAILABILITY_ROOT}/src" "${UWVM_AVAILABILITY_ROOT}/third-parties/fast_io/include")
    target_link_libraries(availability-${model}-contract PRIVATE Threads::Threads)
  endforeach()
endif()
if(UWVM_AVAILABILITY_NATIVE)
  if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23)$")
    message(FATAL_ERROR "Select actual native SDK compiler standard17/20/23")
  endif()
  add_executable(availability-native-contract native_contract.cc)
  target_compile_definitions(availability-native-contract PRIVATE UWVM_ROS_BACKEND=${UWVM_ROS_BACKEND})
  target_include_directories(availability-native-contract PRIVATE "${UWVM_AVAILABILITY_ROOT}/src")
  target_link_libraries(availability-native-contract PRIVATE Threads::Threads)
  if(UWVM_ROS_BACKEND STREQUAL "1")
    if(UWVM_ROS1_INCLUDE_DIRS OR UWVM_ROS1_LIBRARIES)
      if(NOT UWVM_ROS1_INCLUDE_DIRS OR NOT UWVM_ROS1_LIBRARIES)
        message(FATAL_ERROR "Both explicit selected SDK include/library inventories are required")
      endif()
      target_include_directories(availability-native-contract SYSTEM PRIVATE ${UWVM_ROS1_INCLUDE_DIRS})
      target_link_libraries(availability-native-contract PRIVATE ${UWVM_ROS1_LIBRARIES})
    else()
      find_package(catkin REQUIRED COMPONENTS roscpp std_srvs)
      target_include_directories(availability-native-contract SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
      target_link_libraries(availability-native-contract PRIVATE ${catkin_LIBRARIES})
    endif()
  elseif(UWVM_ROS_BACKEND STREQUAL "2")
    find_package(ament_cmake REQUIRED)
    find_package(rclcpp REQUIRED)
    find_package(std_srvs REQUIRED)
    find_package(rmw REQUIRED)
    find_package(rosidl_typesupport_introspection_cpp REQUIRED)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    # Reuse actual selected Gateway compile/link feature results. This directory
    # is a separately pinned producer input, never a guessed distro tag/overlay.
    if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
      message(FATAL_ERROR "Supply actual pinned Gateway feature-header directory")
    endif()
    target_include_directories(availability-native-contract PRIVATE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
    target_link_libraries(availability-native-contract PRIVATE OpenSSL::Crypto)
    if(COMMAND ament_target_dependencies)
      ament_target_dependencies(availability-native-contract rclcpp std_srvs rmw rosidl_typesupport_introspection_cpp)
    else()
      foreach(dependency IN ITEMS rclcpp std_srvs rmw rosidl_typesupport_introspection_cpp)
        if(DEFINED ${dependency}_TARGETS AND NOT "${${dependency}_TARGETS}" STREQUAL "")
          target_link_libraries(availability-native-contract PRIVATE ${${dependency}_TARGETS})
        elseif(TARGET ${dependency}::${dependency})
          target_link_libraries(availability-native-contract PRIVATE ${dependency}::${dependency})
        elseif(DEFINED ${dependency}_LIBRARIES AND NOT "${${dependency}_LIBRARIES}" STREQUAL "")
          target_include_directories(availability-native-contract SYSTEM PRIVATE ${${dependency}_INCLUDE_DIRS})
          target_link_libraries(availability-native-contract PRIVATE ${${dependency}_LIBRARIES})
        else()
          message(FATAL_ERROR "Selected native dependency has no real link interface: ${dependency}")
        endif()
      endforeach()
    endif()
  else()
    message(FATAL_ERROR "Native target needs exact UWVM_ROS_BACKEND1 or2")
  endif()
endif()
