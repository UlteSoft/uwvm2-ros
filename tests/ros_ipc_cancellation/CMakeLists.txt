cmake_minimum_required(VERSION 3.20)
project(uwvm_ipc_native_cancellation LANGUAGES CXX)
set(UWVM_SOURCE_ROOT "" CACHE PATH "Actual complete source root")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Real Linux eventfd/socket/poll required; no substituted platform")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/src/uwvm2/imported/ros/ipc/cancellation_scope.h")
 message(FATAL_ERROR "Actual complete reviewed source root required")
endif()
find_package(Threads REQUIRED)
add_executable(ipc-native-cancellation value_and_linux_contract.cc)
target_compile_features(ipc-native-cancellation PRIVATE cxx_std_17)
target_include_directories(ipc-native-cancellation PRIVATE "${UWVM_SOURCE_ROOT}/src")
target_link_libraries(ipc-native-cancellation PRIVATE Threads::Threads)
