cmake_minimum_required(VERSION 3.20)
project(uwvm_rmw_threaded_contracts LANGUAGES C CXX)
find_package(rmw_uwvm REQUIRED)
find_package(uwvm_rmw_dispatch REQUIRED)
find_package(std_msgs REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(Threads REQUIRED)
get_target_property(_defs rmw_uwvm::rmw_uwvm INTERFACE_COMPILE_DEFINITIONS)
# The shared target's definition is transitive. Actual source additionally
# refuses missing ON through the required diagnostics symbol at link time.
add_executable(uwvm_rmw_threaded_guard guard_contract.c)
target_link_libraries(uwvm_rmw_threaded_guard PRIVATE rmw_uwvm::rmw_uwvm uwvm_rmw_dispatch::uwvm_rmw_dispatch Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_executable(uwvm_rmw_threaded_queue native_queue_model.c)
 target_link_libraries(uwvm_rmw_threaded_queue PRIVATE rmw_uwvm::rmw_uwvm
  std_msgs::std_msgs__rosidl_generator_c std_msgs::std_msgs__rosidl_typesupport_c
  std_msgs::std_msgs__rosidl_typesupport_introspection_c Threads::Threads)
endif()
option(UWVM_BUILD_ACTUAL_THREADED_GUEST "Build with actual SDK33 WASI threads toolchain" OFF)
if(UWVM_BUILD_ACTUAL_THREADED_GUEST)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  message(FATAL_ERROR "Actual Guest example requires the real WASI toolchain")
 endif()
 add_executable(uwvm_rmw_threaded_guest ../examples/guest_guard.c)
 target_link_libraries(uwvm_rmw_threaded_guest PRIVATE rmw_uwvm::rmw_uwvm Threads::Threads)
endif()
option(UWVM_BUILD_ACTUAL_ORIGINAL_GRAPH "Compile actual rclcpp GraphListener API contract" OFF)
if(UWVM_BUILD_ACTUAL_ORIGINAL_GRAPH)
 find_package(rclcpp REQUIRED)
 add_executable(uwvm_rmw_original_graph_contract original_graph_listener.cpp ../examples/original_graph_main.cpp)
 target_compile_features(uwvm_rmw_original_graph_contract PRIVATE cxx_std_17)
 target_link_libraries(uwvm_rmw_original_graph_contract PRIVATE rclcpp::rclcpp rmw_uwvm::rmw_uwvm Threads::Threads)
endif()
foreach(_target uwvm_rmw_threaded_guard uwvm_rmw_threaded_queue)
 if(TARGET ${_target})
  target_compile_options(${_target} PRIVATE -UNDEBUG)
 endif()
endforeach()
# No tests are automatically run. Root must supply true process/CPU/RSS/IO and
# physical-join ownership; no configure-time code executes these candidates.
