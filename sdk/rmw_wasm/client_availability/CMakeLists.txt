cmake_minimum_required(VERSION 3.20)
project(uwvm_original_rmw_client_availability LANGUAGES C CXX)
if(NOT TARGET rmw_uwvm OR NOT TARGET uwvm_rmw_service_transport)
 message(FATAL_ERROR "Actual source-recompiled B/C plus genuine C14 required, no substitutes")
endif()
if(NOT IS_ABSOLUTE "${UWVM_RMW_INCLUDE_DIR}" OR NOT EXISTS "${UWVM_RMW_INCLUDE_DIR}/uwvm_rmw_c/provider.h")
 message(FATAL_ERROR "Exact current overlaid source headers required")
endif()
option(UWVM_RMW_AVAILABILITY_BUILD_MODELS "Build unexecuted local source contracts" OFF)
option(UWVM_RMW_AVAILABILITY_BUILD_C_EXAMPLE "Build full original generated-C client/server main" OFF)
option(UWVM_RMW_AVAILABILITY_BUILD_CPP_EXAMPLE "Build full C API with actual C++ objects companion" OFF)
if(UWVM_RMW_AVAILABILITY_BUILD_MODELS)
 find_package(example_interfaces REQUIRED)
 add_executable(uwvm_rmw_availability_models tests/contract.c "${UWVM_RMW_INCLUDE_DIR}/../src/add_two_ints_codec.c")
 target_compile_features(uwvm_rmw_availability_models PRIVATE c_std_11)
 target_compile_options(uwvm_rmw_availability_models PRIVATE -UNDEBUG)
 target_link_libraries(uwvm_rmw_availability_models PRIVATE rmw_uwvm example_interfaces::example_interfaces__rosidl_generator_c example_interfaces::example_interfaces__rosidl_typesupport_c example_interfaces::example_interfaces__rosidl_typesupport_introspection_c)
 get_property(_cpp TARGET rmw_uwvm PROPERTY UWVM_RMW_CPP_CODECS_ATTACHED)
 if(NOT _cpp OR NOT TARGET uwvm_rmw_subscription_transport)
  message(FATAL_ERROR "Complete real CPP10 and actual C9 needed for separate real-C14 wrapper contract")
 endif()
 add_executable(uwvm_rmw_availability_transport_models tests/transport_contract.cc)
 target_compile_features(uwvm_rmw_availability_transport_models PRIVATE cxx_std_20)
 target_compile_options(uwvm_rmw_availability_transport_models PRIVATE -UNDEBUG)
 target_include_directories(uwvm_rmw_availability_transport_models PRIVATE "${UWVM_RMW_INCLUDE_DIR}/../tests")
 target_link_libraries(uwvm_rmw_availability_transport_models PRIVATE uwvm_rmw_service_transport uwvm_rmw_subscription_transport rmw_uwvm)
endif()
if(UWVM_RMW_AVAILABILITY_BUILD_C_EXAMPLE)
 if(NOT TARGET uwvm_rmw_c_codecs OR NOT TARGET uwvm_rmw_subscription_transport OR NOT TARGET uwvm_rmw_publisher_transport)
  message(FATAL_ERROR "Actual closed full pure-C Registry and providers required; no CPP companion fallback")
 endif()
 add_executable(original_client_c examples/original_client_c.c)
 target_compile_features(original_client_c PRIVATE c_std_11)
 target_link_libraries(original_client_c PRIVATE uwvm_rmw_c_codecs uwvm_rmw_service_transport uwvm_rmw_subscription_transport uwvm_rmw_publisher_transport rmw_uwvm)
endif()
if(UWVM_RMW_AVAILABILITY_BUILD_CPP_EXAMPLE)
 get_property(_cpp TARGET rmw_uwvm PROPERTY UWVM_RMW_CPP_CODECS_ATTACHED)
 if(NOT _cpp)
  message(FATAL_ERROR "Actual original C++ objects/complete Registry must be explicitly attached")
 endif()
 add_library(availability_cpp_objects STATIC examples/native_objects.cpp)
 target_compile_features(availability_cpp_objects PUBLIC cxx_std_20)
 target_link_libraries(availability_cpp_objects PUBLIC rmw_uwvm)
 add_executable(original_client_cpp_objects examples/original_client_cpp_objects.c)
 target_compile_features(original_client_cpp_objects PRIVATE c_std_11)
 target_link_libraries(original_client_cpp_objects PRIVATE availability_cpp_objects uwvm_rmw_service_transport rmw_uwvm)
endif()
