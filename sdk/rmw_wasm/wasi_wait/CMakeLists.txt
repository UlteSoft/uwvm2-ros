cmake_minimum_required(VERSION 3.20)
project(uwvm_rmw_wasi_wait LANGUAGES C)
set(UWVM_RMW_WASI_WAIT_MODE "" CACHE STRING "Explicit STATIC_WASI or NATIVE_UNSUPPORTED")
if(NOT UWVM_RMW_WASI_WAIT_MODE STREQUAL "STATIC_WASI" AND NOT UWVM_RMW_WASI_WAIT_MODE STREQUAL "NATIVE_UNSUPPORTED")
 message(FATAL_ERROR "Choose actual Preview1 Guest or honest native-unsupported contract mode")
endif()
if(BUILD_SHARED_LIBS)
 message(FATAL_ERROR "Process-lifetime Guest wait provider is static only")
endif()
find_package(rmw REQUIRED)
find_package(rcutils REQUIRED)
if(NOT IS_ABSOLUTE "${UWVM_RMW_INCLUDE_DIR}" OR NOT EXISTS "${UWVM_RMW_INCLUDE_DIR}/uwvm_rmw_c/provider.h")
 message(FATAL_ERROR "Actual current source-recompiled original-RMW C profile header required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_ROS_SDK_INCLUDE_DIR}" OR NOT EXISTS "${UWVM_ROS_SDK_INCLUDE_DIR}/roswasm/roswasmc.h")
 message(FATAL_ERROR "Actual reviewed C SDK headers required, no facade headers")
endif()
include(cmake/Probe.cmake)
uwvm_rmw_probe_wasi_wait("${CMAKE_CURRENT_BINARY_DIR}/actual-wasi-features")
if(UWVM_RMW_WASI_WAIT_MODE STREQUAL "STATIC_WASI" AND NOT UWVM_RMW_WASI_WAIT_P1_API)
 message(FATAL_ERROR "Actual Preview1 header/import function failed real compile/link")
endif()
if(UWVM_RMW_WASI_WAIT_MODE STREQUAL "NATIVE_UNSUPPORTED" AND UWVM_RMW_WASI_WAIT_P1_API)
 message(FATAL_ERROR "Native-unsupported mode cannot substitute an actual Guest build")
endif()
add_library(uwvm_rmw_wasi_wait STATIC src/wasi_wait.c)
target_compile_features(uwvm_rmw_wasi_wait PUBLIC c_std_11)
target_include_directories(uwvm_rmw_wasi_wait PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${UWVM_RMW_INCLUDE_DIR}" "${UWVM_ROS_SDK_INCLUDE_DIR}"
 PRIVATE "${UWVM_RMW_WASI_WAIT_FEATURE_INCLUDE_DIR}")
target_link_libraries(uwvm_rmw_wasi_wait PUBLIC rmw::rmw rcutils::rcutils)
option(UWVM_RMW_WASI_WAIT_BUILD_CONTRACTS "Build only; never execute authored source contracts" OFF)
if(UWVM_RMW_WASI_WAIT_BUILD_CONTRACTS)
 if(UWVM_RMW_WASI_WAIT_MODE STREQUAL "STATIC_WASI")
  add_executable(wasi_wait_values tests/value_contract.c)
  add_executable(wasi_wait_guest tests/guest_contract.c)
  foreach(_target wasi_wait_values wasi_wait_guest)
   target_include_directories(${_target} PRIVATE "${UWVM_RMW_WASI_WAIT_FEATURE_INCLUDE_DIR}")
   target_link_libraries(${_target} PRIVATE uwvm_rmw_wasi_wait)
  endforeach()
 else()
  add_executable(wasi_wait_native_unsupported tests/native_unsupported.c)
  target_link_libraries(wasi_wait_native_unsupported PRIVATE uwvm_rmw_wasi_wait)
 endif()
endif()
option(UWVM_RMW_WASI_WAIT_BUILD_COMPOSITE "Build complete original-RMW actual transport/bootstrap example" OFF)
if(UWVM_RMW_WASI_WAIT_BUILD_COMPOSITE)
 if(NOT UWVM_RMW_WASI_WAIT_MODE STREQUAL "STATIC_WASI")
  message(FATAL_ERROR "Full composition is an actual Guest artifact")
 endif()
 foreach(_needed rmw_uwvm uwvm_rmw_c_codecs uwvm_rmw_service_transport uwvm_rmw_subscription_transport uwvm_rmw_publisher_transport)
  if(NOT TARGET "${_needed}")
   message(FATAL_ERROR "Actual complete reviewed ${_needed} target required; no source stub")
  endif()
 endforeach()
 if(TARGET uwvm_rmw_platform::uwvm_rmw_platform)
  set(_platform uwvm_rmw_platform::uwvm_rmw_platform)
 elseif(TARGET uwvm_rmw_platform)
  set(_platform uwvm_rmw_platform)
 else()
  message(FATAL_ERROR "Actual prepared A platform target required")
 endif()
 add_executable(wasi_wait_original_rmw examples/original_rmw_composite.c)
 target_link_libraries(wasi_wait_original_rmw PRIVATE uwvm_rmw_wasi_wait rmw_uwvm uwvm_rmw_c_codecs
  uwvm_rmw_service_transport uwvm_rmw_subscription_transport uwvm_rmw_publisher_transport ${_platform})
endif()
# No add_test/try_run/execute_process, Guest instance, native actor or network
# action is automatically invoked. Root later owns all actual bounded commands.
