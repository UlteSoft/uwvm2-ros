cmake_minimum_required(VERSION 3.20)
project(uwvm_wasi_signal_contracts LANGUAGES CXX)
find_package(uwvm_rmw_threaded_platform REQUIRED)
set(_actual uwvm_rmw_threaded_platform::uwvm_rmw_wasi_signal)
get_target_property(_feature "${_actual}" UWVM_WASI_SIGNAL_ADMITTED)
option(UWVM_SIGNAL_RECEIVER_GUEST "Build actual WASI pipe receiver witness" OFF)
option(UWVM_SIGNAL_BUILD_ORIGINAL_REENTRY "Build genuine original callback reentry contract body" OFF)
if(UWVM_SIGNAL_RECEIVER_GUEST)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI" OR NOT _feature STREQUAL "1")
    message(FATAL_ERROR "Actual admitted WASI signal library required")
  endif()
  foreach(_role UWVM_SIGNAL_READ_FD UWVM_SIGNAL_SOURCE_COOKIE)
    if(NOT DEFINED ${_role} OR NOT "${${_role}}" MATCHES "^[0-9]+$")
      message(FATAL_ERROR "Supply actual immutable numeric role: ${_role}")
    endif()
  endforeach()
  add_executable(receiver_guest receiver_guest.cpp)
  target_link_libraries(receiver_guest PRIVATE "${_actual}")
  target_compile_definitions(receiver_guest PRIVATE
    UWVM_SIGNAL_READ_FD=${UWVM_SIGNAL_READ_FD}
    UWVM_SIGNAL_SOURCE_COOKIE=${UWVM_SIGNAL_SOURCE_COOKIE}ULL)
  foreach(_expected UWVM_SIGNAL_EXPECTED_ERROR UWVM_SIGNAL_EXPECTED_SIGINT UWVM_SIGNAL_EXPECTED_SIGTERM UWVM_SIGNAL_FRAME_LIMIT UWVM_SIGNAL_STOP_BEFORE_ACTIVATE)
    if(DEFINED ${_expected})
      if(NOT "${${_expected}}" MATCHES "^[0-9]+$")
        message(FATAL_ERROR "Bad finite expected contract parameter: ${_expected}")
      endif()
      target_compile_definitions(receiver_guest PRIVATE ${_expected}=${${_expected}})
    endif()
  endforeach()
elseif(NOT UWVM_SIGNAL_BUILD_ORIGINAL_REENTRY)
  if(_feature)
    message(FATAL_ERROR "The refusal executable must use an actual feature-zero library")
  endif()
  add_executable(refusal refusal.cpp)
  target_link_libraries(refusal PRIVATE "${_actual}")
  if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    target_compile_definitions(refusal PRIVATE _WASI_EMULATED_SIGNAL)
  endif()
endif()

if(UWVM_SIGNAL_BUILD_ORIGINAL_REENTRY)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI" OR NOT _feature STREQUAL "1")
    message(FATAL_ERROR "Actual admitted original/WASI target is required")
  endif()
  find_package(rclcpp REQUIRED)
  add_library(original_worker_reentry STATIC original_worker_reentry.cpp)
  target_compile_features(original_worker_reentry PUBLIC cxx_std_17)
  target_link_libraries(original_worker_reentry PUBLIC rclcpp::rclcpp "${_actual}")
endif()
