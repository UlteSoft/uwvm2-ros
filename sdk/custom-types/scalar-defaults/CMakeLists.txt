# Explicit actual native SDK component; configuration/build does not run it.
cmake_minimum_required(VERSION 3.20)
project(uwvm_scalar_defaults_native LANGUAGES CXX)
option(UWVM_BUILD_SCALAR_DEFAULTS_NATIVE "Build actual selected xArm generated-type contract" OFF)
if(NOT UWVM_BUILD_SCALAR_DEFAULTS_NATIVE)
 message(FATAL_ERROR "Select the actual original SDK component explicitly")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SCALAR_DEFAULTS_BUNDLE}" OR NOT EXISTS "${UWVM_SCALAR_DEFAULTS_BUNDLE}/native_ros2.hpp")
 message(FATAL_ERROR "Actual complete independently bound generated bundle required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SCALAR_DEFAULTS_SOURCE}" OR NOT EXISTS "${UWVM_SCALAR_DEFAULTS_SOURCE}/src/uwvm2/imported/ros/ipc/custom_type.h")
 message(FATAL_ERROR "Exact reviewed full source root required")
endif()
find_package(ament_cmake REQUIRED)
find_package(xarm_msgs REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
find_package(service_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
add_executable(scalar_defaults_native native_xarm_contract.cc)
target_compile_features(scalar_defaults_native PRIVATE cxx_std_17)
target_include_directories(scalar_defaults_native PRIVATE "${UWVM_SCALAR_DEFAULTS_BUNDLE}" "${UWVM_SCALAR_DEFAULTS_SOURCE}/src" "${UWVM_SCALAR_DEFAULTS_SOURCE}/sdk/include")
ament_target_dependencies(scalar_defaults_native xarm_msgs rosidl_typesupport_cpp rosidl_typesupport_introspection_cpp service_msgs builtin_interfaces)
# No try_run/add_test/execute_process/native peer launch or default grant.
