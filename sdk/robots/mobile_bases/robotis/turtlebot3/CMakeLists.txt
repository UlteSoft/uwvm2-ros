cmake_minimum_required(VERSION 3.20)
project(uwvm_turtlebot3 LANGUAGES C CXX)
# Explicit opt-in targets: adding this directory to a superbuild does no work.
option(UWVM_TURTLEBOT3_BUILD_CONTRACTS "Build authored whole-value contracts" OFF)
option(UWVM_TURTLEBOT3_BUILD_WASI_EXAMPLES "Build genuine WASI C/C++ entries" OFF)
set(UWVM_TURTLEBOT3_PROFILE "" CACHE STRING "Exactly noetic, humble, jazzy or main")
set_property(CACHE UWVM_TURTLEBOT3_PROFILE PROPERTY STRINGS noetic humble jazzy main)
set(UWVM_TURTLEBOT3_SOURCE_ROOT "" CACHE PATH "Absolute actual source snapshot")
set(UWVM_TURTLEBOT3_GENERATED "" CACHE PATH "Absolute exact matching generated family")
if(NOT UWVM_TURTLEBOT3_BUILD_CONTRACTS AND NOT UWVM_TURTLEBOT3_BUILD_WASI_EXAMPLES)
  return()
endif()
if(NOT UWVM_TURTLEBOT3_PROFILE MATCHES "^(noetic|humble|jazzy|main)$")
  message(FATAL_ERROR "One explicit original TurtleBot3 profile required")
endif()
foreach(path IN ITEMS "${UWVM_TURTLEBOT3_SOURCE_ROOT}" "${UWVM_TURTLEBOT3_GENERATED}")
  if(NOT IS_ABSOLUTE "${path}" OR NOT IS_DIRECTORY "${path}")
    message(FATAL_ERROR "Actual absolute source and generated directories required")
  endif()
endforeach()
set(original "${CMAKE_CURRENT_SOURCE_DIR}/catalogs/${UWVM_TURTLEBOT3_PROFILE}")
set(bundle "${UWVM_TURTLEBOT3_GENERATED}")
foreach(name IN ITEMS generation.vendor.json preparation.vendor.json profile-selection.json)
  if(IS_SYMLINK "${bundle}/${name}")
    message(FATAL_ERROR "Symlinked bundle metadata refused")
  endif()
endforeach()
file(READ "${bundle}/generation.vendor.json" generation)
file(READ "${bundle}/preparation.vendor.json" preparation)
file(READ "${bundle}/profile-selection.json" selection)
file(READ "${original}/source-lock.json" lock)
file(SHA256 "${original}/source-lock.json" lock_sha)
string(JSON captured_lock GET "${selection}" source_lock_sha256)
string(JSON captured_profile GET "${selection}" profile)
if(NOT captured_lock STREQUAL lock_sha OR NOT captured_profile STREQUAL UWVM_TURTLEBOT3_PROFILE)
  message(FATAL_ERROR "Exact message/driver/source profile differs from bundle")
endif()
file(SHA256 "${original}/catalog.json" catalog_sha)
string(JSON captured_catalog GET "${preparation}" original_catalog sha256)
if(NOT captured_catalog STREQUAL catalog_sha)
  message(FATAL_ERROR "Original complete catalog differs from bundle")
endif()
# Verify definition bytes as well as catalog paths. Identical catalog schemas
# can refer to different Humble/Jazzy/main original bodies and defaults.
string(JSON count LENGTH "${lock}" definitions)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON name GET "${lock}" definitions ${index} generation_file)
  string(JSON expected_sha GET "${lock}" definitions ${index} sha256)
  string(JSON prepared_sha GET "${preparation}" original_definitions "${name}" sha256)
  file(SHA256 "${original}/${name}" actual_sha)
  if(NOT expected_sha STREQUAL actual_sha OR NOT prepared_sha STREQUAL actual_sha)
    message(FATAL_ERROR "Complete original definition differs: ${name}")
  endif()
endforeach()
string(JSON count LENGTH "${generation}" outputs)
string(JSON native GET "${generation}" native_registry_emitted)
string(JSON mode GET "${preparation}" mode)
if(native)
  if(NOT count EQUAL 16 OR NOT mode STREQUAL "bind")
    message(FATAL_ERROR "Native products require the actual independently bound complete bundle")
  endif()
  string(JSON native_manifest GET "${generation}" outputs "manifest.hpp" sha256)
  string(JSON version GET "${selection}" ros_version)
  string(JSON native_adapter GET "${generation}" outputs "native_ros${version}.hpp" sha256)
elseif(NOT count EQUAL 14 OR NOT mode STREQUAL "prepare")
  message(FATAL_ERROR "All six complete record and interface products required")
endif()
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON name MEMBER "${generation}" outputs ${index})
  if(name MATCHES "(^/|(^|/)[.][.](/|$))" OR IS_SYMLINK "${bundle}/${name}")
    message(FATAL_ERROR "Generated product escaped selected family")
  endif()
  string(JSON expected_sha GET "${generation}" outputs "${name}" sha256)
  string(JSON selected_sha GET "${selection}" products "${name}" sha256)
  string(JSON expected_bytes GET "${generation}" outputs "${name}" bytes)
  file(SHA256 "${bundle}/${name}" actual_sha)
  file(SIZE "${bundle}/${name}" actual_bytes)
  if(NOT actual_sha STREQUAL expected_sha OR NOT actual_sha STREQUAL selected_sha OR NOT actual_bytes EQUAL expected_bytes)
    message(FATAL_ERROR "Exact generated product differs: ${name}")
  endif()
endforeach()
set(profiles noetic humble jazzy main)
list(FIND profiles "${UWVM_TURTLEBOT3_PROFILE}" profile_index)
math(EXPR profile_number "${profile_index}+1")
add_library(turtlebot3_selected INTERFACE)
target_include_directories(turtlebot3_selected INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${bundle}" "${UWVM_TURTLEBOT3_SOURCE_ROOT}/sdk/include")
target_compile_definitions(turtlebot3_selected INTERFACE "UWVM_TURTLEBOT3_PROFILE=${profile_number}")
if(UWVM_TURTLEBOT3_BUILD_CONTRACTS)
  add_executable(turtlebot3_contract_c "tests/contract_${UWVM_TURTLEBOT3_PROFILE}.c")
  add_executable(turtlebot3_contract_cpp "tests/contract_${UWVM_TURTLEBOT3_PROFILE}.cc")
  target_compile_options(turtlebot3_contract_c PRIVATE -UNDEBUG)
  target_compile_options(turtlebot3_contract_cpp PRIVATE -UNDEBUG)
  set_property(TARGET turtlebot3_contract_c PROPERTY C_STANDARD 11)
  set_property(TARGET turtlebot3_contract_cpp PROPERTY CXX_STANDARD 20)
  target_link_libraries(turtlebot3_contract_c PRIVATE turtlebot3_selected)
  target_link_libraries(turtlebot3_contract_cpp PRIVATE turtlebot3_selected)
endif()
if(UWVM_TURTLEBOT3_BUILD_WASI_EXAMPLES)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
    message(FATAL_ERROR "Actual WASI toolchain required for Guest entries")
  endif()
  add_executable(turtlebot3_guest_c examples/guest.c)
  add_executable(turtlebot3_guest_cpp examples/guest.cc)
  target_link_libraries(turtlebot3_guest_c PRIVATE turtlebot3_selected)
  target_link_libraries(turtlebot3_guest_cpp PRIVATE turtlebot3_selected)
  set_property(TARGET turtlebot3_guest_c PROPERTY C_STANDARD 11)
  set_property(TARGET turtlebot3_guest_cpp PROPERTY CXX_STANDARD 20)
endif()
foreach(target IN ITEMS turtlebot3_contract_c turtlebot3_contract_cpp turtlebot3_guest_c turtlebot3_guest_cpp)
  if(TARGET ${target} AND CMAKE_SYSTEM_NAME STREQUAL "WASI")
    set_target_properties(${target} PROPERTIES SUFFIX ".wasm")
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>")
    target_link_options(${target} PRIVATE "-Wl,--threads=1")
  endif()
endforeach()
