cmake_minimum_required(VERSION 3.24)
project(uwvm_agilex_scout LANGUAGES C CXX)
option(UWVM_SCOUT_BUILD_CONTRACTS "Build actual C/C++ complete-value contracts; configure never runs them" OFF)
option(UWVM_SCOUT_BUILD_WASI_EXAMPLES "Build real default-denied C/C++ WASI applications" OFF)
set(UWVM_SCOUT_ROS_VERSION "" CACHE STRING "One explicit ROS major, 1 or 2")
set(UWVM_SCOUT_SOURCE_ROOT "" CACHE PATH "Actual paired uwvm2-ros SDK source root")
set(UWVM_SCOUT_GENERATED "" CACHE PATH "One matching complete generated SCOUT family")
if(NOT UWVM_SCOUT_BUILD_CONTRACTS AND NOT UWVM_SCOUT_BUILD_WASI_EXAMPLES)
  return()
endif()
if(NOT UWVM_SCOUT_ROS_VERSION MATCHES "^[12]$" OR
   NOT EXISTS "${UWVM_SCOUT_SOURCE_ROOT}/sdk/include/roswasm/roswasmc.h" OR
   NOT EXISTS "${UWVM_SCOUT_GENERATED}/generation.vendor.json")
  message(FATAL_ERROR "Explicit ROS major, actual core SDK and whole generated SCOUT family are required")
endif()
file(READ "${UWVM_SCOUT_GENERATED}/generation.vendor.json" _generation)
string(JSON _records LENGTH "${_generation}" records)
math(EXPR _last "${_records}-1")
foreach(_i RANGE 0 ${_last})
  string(JSON _major GET "${_generation}" records ${_i} ros_version)
  if(NOT _major STREQUAL UWVM_SCOUT_ROS_VERSION)
    message(FATAL_ERROR "Selected generated family has the wrong ROS major")
  endif()
endforeach()
# Check every exact product, including the native exporter source. This is an
# immutable source-selection check, not Native type binding or authorization.
string(JSON _count LENGTH "${_generation}" outputs)
math(EXPR _last "${_count}-1")
foreach(_i RANGE 0 ${_last})
  string(JSON _name MEMBER "${_generation}" outputs ${_i})
  string(JSON _expected GET "${_generation}" outputs "${_name}" sha256)
  file(SHA256 "${UWVM_SCOUT_GENERATED}/${_name}" _actual)
  if(NOT _actual STREQUAL _expected)
    message(FATAL_ERROR "Generated SCOUT source differs: ${_name}")
  endif()
endforeach()
add_library(uwvm_scout_sdk INTERFACE)
target_include_directories(uwvm_scout_sdk INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${UWVM_SCOUT_SOURCE_ROOT}/sdk/include" "${UWVM_SCOUT_GENERATED}")
target_compile_definitions(uwvm_scout_sdk INTERFACE UWVM_SCOUT_ROS_VERSION=${UWVM_SCOUT_ROS_VERSION})
if(UWVM_SCOUT_BUILD_CONTRACTS)
  add_executable(scout_contract_c tests/contract.c)
  add_executable(scout_contract_cpp tests/contract.cc)
  foreach(_target scout_contract_c scout_contract_cpp)
    target_link_libraries(${_target} PRIVATE uwvm_scout_sdk)
    if(MSVC)
      target_compile_options(${_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
if(UWVM_SCOUT_BUILD_WASI_EXAMPLES)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
    message(FATAL_ERROR "Real WASI toolchain required for SCOUT import examples")
  endif()
  add_executable(scout_guest_c examples/guest.c)
  add_executable(scout_guest_cpp examples/guest.cc)
  target_link_libraries(scout_guest_c PRIVATE uwvm_scout_sdk)
  target_link_libraries(scout_guest_cpp PRIVATE uwvm_scout_sdk)
endif()
foreach(_target scout_contract_c scout_contract_cpp scout_guest_c scout_guest_cpp)
  if(TARGET ${_target})
    target_compile_features(${_target} PRIVATE c_std_11 cxx_std_20)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${_target} PRIVATE -ffunction-sections -fdata-sections)
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
      set_target_properties(${_target} PROPERTIES SUFFIX ".wasm")
      target_compile_options(${_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>")
      # A one-job build does not constrain wasm-ld's own worker pool. Keep this
      # bounded SDK lane serial inside the shared operator cgroup as well.
      target_link_options(${_target} PRIVATE "-Wl,--threads=1")
    endif()
  endif()
endforeach()
