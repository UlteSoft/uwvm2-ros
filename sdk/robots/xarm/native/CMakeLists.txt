cmake_minimum_required(VERSION 3.16)
project(uwvm_xarm_robot_msg_native_oracles LANGUAGES CXX)
# This entry only declares source targets. It never downloads, generates types,
# creates a robot connection, registers tests or starts an executable.
set(UWVM_SOURCE "" CACHE PATH "Actual admitted complete current source")
set(UWVM_XARM_GENERATED "" CACHE PATH "Actual whole native-bound generated bundle")
set(UWVM_CUSTOM_NATIVE_ROS_VERSION "" CACHE STRING "Exactly 1 or 2")
set(BUILD_TESTING OFF CACHE BOOL "No automatic execution" FORCE)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/src/uwvm2/imported/ros/ipc/custom_type.h")
  message(FATAL_ERROR "Actual current complete source required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_XARM_GENERATED}" OR NOT EXISTS "${UWVM_XARM_GENERATED}/generation.json" OR NOT EXISTS "${UWVM_XARM_GENERATED}/manifest.hpp" OR NOT EXISTS "${UWVM_XARM_GENERATED}/native_ros${UWVM_CUSTOM_NATIVE_ROS_VERSION}.hpp" OR "${UWVM_XARM_GENERATED}" MATCHES "[;\"\\\r\n]")
  message(FATAL_ERROR "Actual complete native-bound generation required")
endif()
add_executable(xarm-robot-msg-native-roundtrip roundtrip.cc)
add_executable(xarm-robot-msg-native-publisher publisher.cc)
foreach(target xarm-robot-msg-native-roundtrip xarm-robot-msg-native-publisher)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=${UWVM_CUSTOM_NATIVE_ROS_VERSION})
  target_include_directories(${target} PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/src" "${UWVM_XARM_GENERATED}")
  if(target STREQUAL "xarm-robot-msg-native-roundtrip")
    set(native_file roundtrip.cc)
  else()
    set(native_file publisher.cc)
  endif()
  target_compile_options(${target} PRIVATE -O1 -g0 -UNDEBUG -std=gnu++17 -MD -MF "${CMAKE_BINARY_DIR}/CMakeFiles/${target}.dir/${native_file}.o.d" -MT "${CMAKE_BINARY_DIR}/CMakeFiles/${target}.dir/${native_file}.o")
  target_link_options(${target} PRIVATE -Wl,-t)
endforeach()
if(UWVM_CUSTOM_NATIVE_ROS_VERSION STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS roscpp std_msgs xarm_msgs)
  foreach(target xarm-robot-msg-native-roundtrip xarm-robot-msg-native-publisher)
    target_include_directories(${target} PRIVATE ${catkin_INCLUDE_DIRS})
    target_link_libraries(${target} PRIVATE ${catkin_LIBRARIES})
  endforeach()
elseif(UWVM_CUSTOM_NATIVE_ROS_VERSION STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(builtin_interfaces REQUIRED)
  find_package(xarm_msgs REQUIRED)
  find_package(rosidl_typesupport_cpp REQUIRED)
  foreach(target xarm-robot-msg-native-roundtrip xarm-robot-msg-native-publisher)
    ament_target_dependencies(${target} PUBLIC rclcpp std_msgs builtin_interfaces xarm_msgs rosidl_typesupport_cpp)
  endforeach()
else()
  message(FATAL_ERROR "Exact actual native ROS major required")
endif()
