cmake_minimum_required(VERSION 3.16)
project(uwvm_mobile_original_native_witness LANGUAGES CXX)
option(UWVM_MOBILE_BUILD_NATIVE_WITNESS "Explicitly build the original ROS2 native witness" OFF)
if(NOT UWVM_MOBILE_BUILD_NATIVE_WITNESS)
  return()
endif()
if(NOT EXISTS "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include/fast_io.h")
  message(FATAL_ERROR "Actual UWVM source root with vendored fast_io is required")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
add_executable(uwvm_mobile_native_fjt_peer fjt_peer.cc)
target_compile_features(uwvm_mobile_native_fjt_peer PRIVATE cxx_std_20)
target_include_directories(uwvm_mobile_native_fjt_peer PRIVATE "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include")
ament_target_dependencies(uwvm_mobile_native_fjt_peer rclcpp rclcpp_action control_msgs trajectory_msgs builtin_interfaces)
# Real executable final link, not a cached STATIC_LIBRARY pseudo-probe. All
# actual compiler/SDK/type-support/generator inputs still require qualification.
