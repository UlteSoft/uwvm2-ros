cmake_minimum_required(VERSION 3.16)
project(uwvm_mobile_wasi_examples LANGUAGES C CXX)
option(UWVM_MOBILE_BUILD_GUESTS "Explicitly build complete typed WASI guests" OFF)
option(UWVM_MOBILE_BUILD_FJT_GUESTS "Build only the Kinova/Stretch FJT WASI examples" OFF)
option(UWVM_MOBILE_BUILD_SPOT_GUESTS "Build only the Spot bridge WASI example" OFF)
# Preserve the original explicit aggregate switch, while allowing a deployment
# to omit every source and generated dependency of the unselected family.
if(UWVM_MOBILE_BUILD_GUESTS)
  set(UWVM_MOBILE_BUILD_FJT_GUESTS ON)
  set(UWVM_MOBILE_BUILD_SPOT_GUESTS ON)
endif()
if(NOT UWVM_MOBILE_BUILD_FJT_GUESTS AND NOT UWVM_MOBILE_BUILD_SPOT_GUESTS)
  return()
endif()
if(NOT CMAKE_C_COMPILER_TARGET MATCHES "^wasm32(-unknown)?-(wasi|wasip1)$")
  message(FATAL_ERROR "An actual explicit WASI compiler target is required; host compilation is not Guest proof")
endif()
if(NOT EXISTS "${UWVM_SOURCE_ROOT}/sdk/include/roswasm/ros2.h")
  message(FATAL_ERROR "Actual SDK source root is required")
endif()
if(UWVM_MOBILE_BUILD_FJT_GUESTS)
  if(NOT EXISTS "${UWVM_MOBILE_FJT_PRODUCT}/bounded_records.h" OR NOT EXISTS "${UWVM_MOBILE_FJT_PRODUCT}/bounded_records.hpp")
    message(FATAL_ERROR "UWVM_MOBILE_FJT_PRODUCT must name an actual frozen generated product")
  endif()
  add_executable(uwvm_mobile_fjt_guest examples/fjt_guest.c)
  set_target_properties(uwvm_mobile_fjt_guest PROPERTIES SUFFIX ".wasm")
  target_compile_features(uwvm_mobile_fjt_guest PRIVATE c_std_11)
  target_include_directories(uwvm_mobile_fjt_guest PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_MOBILE_FJT_PRODUCT}")
  # Real complete generated C and C++ record contracts, separately named.
  add_executable(uwvm_mobile_generated_c_contract tests/generated_contract.c)
  set_target_properties(uwvm_mobile_generated_c_contract PROPERTIES SUFFIX ".wasm")
  target_compile_features(uwvm_mobile_generated_c_contract PRIVATE c_std_11)
  target_include_directories(uwvm_mobile_generated_c_contract PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_MOBILE_FJT_PRODUCT}")
  add_executable(uwvm_mobile_generated_cpp_contract tests/generated_contract.cc)
  set_target_properties(uwvm_mobile_generated_cpp_contract PROPERTIES SUFFIX ".wasm")
  target_compile_features(uwvm_mobile_generated_cpp_contract PRIVATE cxx_std_20)
  # The ordinary WASI SDK libc++ profile is built without C++ exception support.
  # ROS operation refusals remain numeric; allocation failure follows the selected
  # standard library's termination policy, as documented by roswasmcpp.hpp.
  target_compile_options(uwvm_mobile_generated_cpp_contract PRIVATE -fno-exceptions)
  target_include_directories(uwvm_mobile_generated_cpp_contract PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_MOBILE_FJT_PRODUCT}")
endif()
if(UWVM_MOBILE_BUILD_SPOT_GUESTS)
  if(NOT EXISTS "${UWVM_MOBILE_SPOT_PRODUCT}/bounded_records.h" OR NOT EXISTS "${UWVM_MOBILE_SPOT_PRODUCT}/bounded_records.hpp")
    message(FATAL_ERROR "UWVM_MOBILE_SPOT_PRODUCT must name an actual frozen generated product")
  endif()
  add_executable(uwvm_spot_typed_guest spot/examples/guest_sequence.c)
  set_target_properties(uwvm_spot_typed_guest PROPERTIES SUFFIX ".wasm")
  target_compile_features(uwvm_spot_typed_guest PRIVATE c_std_11)
  target_include_directories(uwvm_spot_typed_guest PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_MOBILE_SPOT_PRODUCT}")
endif()
# Toolchain owns libc/libc++, memory/stack, import permissions and final validate.
# No -allow-undefined, fake Host exports, test execution or capability grants.
