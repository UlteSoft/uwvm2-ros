cmake_minimum_required(VERSION 3.24)
project(uwvm_spot_full_protobuf LANGUAGES C CXX)
option(UWVM_SPOT_PROTOBUF_C "Build actual complete nanopb C field SDK" OFF)
option(UWVM_SPOT_PROTOBUF_CPP "Build actual complete protobuf Lite C++ field SDK" OFF)
option(UWVM_SPOT_PROTOBUF_CONTRACTS "Build real generated-type contract executables; never run at configure" OFF)
option(UWVM_SPOT_PROTOBUF_ROS_GUESTS "Build actual typed State ROS Guest entries" OFF)
set(UWVM_ROS_SOURCE_ROOT "" CACHE PATH "Actual source root for finite symbol-resolution oracle")
set(UWVM_SPOT_BRIDGE_PRODUCT "${CMAKE_CURRENT_SOURCE_DIR}/bridge-generated" CACHE PATH
  "One coherent complete Spot bridge generation; never a Native identity or permission preset")
if(NOT UWVM_SPOT_PROTOBUF_C AND NOT UWVM_SPOT_PROTOBUF_CPP)
  return()
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/GeneratedSources.cmake")
# These sources are checked-in results of actual pinned protoc/nanopb execution.
# Configure never generates them or imports a Python module.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/generated/source-products.json" _products)
string(JSON _count LENGTH "${_products}" products)
math(EXPR _last "${_count}-1")
foreach(_i RANGE 0 ${_last})
  string(JSON _name GET "${_products}" products ${_i} path)
  string(JSON _expected GET "${_products}" products ${_i} sha256)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/generated/${_name}" _actual)
  if(NOT _actual STREQUAL _expected)
    message(FATAL_ERROR "Actual generated full SDK source differs: ${_name}")
  endif()
endforeach()
add_library(uwvm_spot_proto_wire STATIC src/wire.c generated/schema.c)
target_compile_features(uwvm_spot_proto_wire PUBLIC c_std_11)
target_include_directories(uwvm_spot_proto_wire PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(UWVM_SPOT_PROTOBUF_C)
  add_library(uwvm_spot_nanopb STATIC src/document.c generated/nanopb_types.c
    upstream/nanopb/pb_common.c src/nanopb_decode_merged.c upstream/nanopb/pb_encode.c
    ${UWVM_SPOT_NANOPB_GENERATED})
  target_compile_features(uwvm_spot_nanopb PUBLIC c_std_11)
  target_include_directories(uwvm_spot_nanopb PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/generated/c"
    "${CMAKE_CURRENT_SOURCE_DIR}/upstream/nanopb")
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(uwvm_spot_nanopb PUBLIC
      "-include${CMAKE_CURRENT_SOURCE_DIR}/include/uwvm_spot_pb_config.h")
  elseif(MSVC)
    target_compile_options(uwvm_spot_nanopb PUBLIC
      "/FI${CMAKE_CURRENT_SOURCE_DIR}/include/uwvm_spot_pb_config.h")
  else()
    message(FATAL_ERROR "No verified forced nanopb layout/config include contract for this compiler")
  endif()
  target_link_libraries(uwvm_spot_nanopb PUBLIC uwvm_spot_proto_wire)
  add_library(uwvm_spot_c_example OBJECT examples/inspect_state.c)
  target_link_libraries(uwvm_spot_c_example PRIVATE uwvm_spot_nanopb)
endif()
if(UWVM_SPOT_PROTOBUF_CPP)
  # Original runtime uses genuine std::mutex/call_once. Do not patch them quiet
  # or pretend a legacy single-thread WASI SDK supplies those implementations.
  if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    if(NOT CMAKE_CXX_COMPILER_TARGET MATCHES "wasm32-wasi-threads")
      message(FATAL_ERROR "C++ Lite WASI lane requires actual pthread libc++ and admitted threaded Runtime")
    endif()
    if(NOT EXISTS "${CMAKE_TOOLCHAIN_FILE}")
      message(FATAL_ERROR "Explicit actual SDK33 Threads+EH toolchain source is required")
    endif()
    file(SHA256 "${CMAKE_TOOLCHAIN_FILE}" _toolchain_sha)
    if(NOT _toolchain_sha STREQUAL "417c102556d8375b331e0dbbd69856c2d29d2222957ff34dfe266c7abebc7730")
      message(FATAL_ERROR "This bounded WASI recipe requires the actual reviewed SDK33 threaded-EH toolchain source")
    endif()
  endif()
  find_package(Threads REQUIRED)
  add_library(uwvm_spot_lite_runtime STATIC ${UWVM_SPOT_PROTOBUF_LITE_RUNTIME})
  target_compile_features(uwvm_spot_lite_runtime PUBLIC cxx_std_17)
  target_include_directories(uwvm_spot_lite_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/upstream/protobuf/src")
  target_link_libraries(uwvm_spot_lite_runtime PUBLIC Threads::Threads)
  add_library(uwvm_spot_protobuf_lite STATIC ${UWVM_SPOT_CPP_GENERATED})
  target_compile_features(uwvm_spot_protobuf_lite PUBLIC cxx_std_17)
  target_include_directories(uwvm_spot_protobuf_lite PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/generated/cpp")
  target_link_libraries(uwvm_spot_protobuf_lite PUBLIC uwvm_spot_lite_runtime uwvm_spot_proto_wire)
  add_library(uwvm_spot_cpp_example OBJECT examples/inspect_state.cc)
  target_link_libraries(uwvm_spot_cpp_example PRIVATE uwvm_spot_protobuf_lite)
endif()
if(UWVM_SPOT_PROTOBUF_CONTRACTS)
  if(NOT EXISTS "${UWVM_ROS_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
    message(FATAL_ERROR "Explicit actual source symbol-oracle dependency required")
  endif()
  include("${UWVM_ROS_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
  if(UWVM_SPOT_PROTOBUF_C)
    add_executable(spot_proto_c_contract tests/document_contract.c)
    target_link_libraries(spot_proto_c_contract PRIVATE uwvm_spot_nanopb)
    # Keep this genuinely C executable's driver inputs explicit as well as
    # the shared target-closure and linker checks. These do not alter flags.
    foreach(_c_role CMAKE_C_FLAGS CMAKE_C_STANDARD_LIBRARIES CMAKE_C_LINK_FLAGS CMAKE_C_LINK_EXECUTABLE)
      if(DEFINED ${_c_role})
        uwvm_current_native_link_mode_guard("${${_c_role}}")
      endif()
    endforeach()
    foreach(_c_config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL ${CMAKE_CONFIGURATION_TYPES} ${CMAKE_BUILD_TYPE})
      string(TOUPPER "${_c_config}" _c_upper)
      uwvm_current_native_link_mode_guard("${CMAKE_C_FLAGS_${_c_upper}}")
    endforeach()
    uwvm_current_native_link_environment(spot_proto_c_contract)
  endif()
  if(UWVM_SPOT_PROTOBUF_CPP)
    add_executable(spot_proto_cpp_contract tests/document_contract.cc)
    target_link_libraries(spot_proto_cpp_contract PRIVATE uwvm_spot_protobuf_lite)
    uwvm_current_native_link_environment(spot_proto_cpp_contract Threads::Threads)
  endif()
  foreach(_target spot_proto_c_contract spot_proto_cpp_contract)
    if(TARGET ${_target})
      if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
        set_target_properties(${_target} PROPERTIES SUFFIX ".wasm")
      endif()
      if(MSVC)
        target_compile_options(${_target} PRIVATE /UNDEBUG)
      else()
        target_compile_options(${_target} PRIVATE -UNDEBUG)
      endif()
    endif()
  endforeach()
endif()
