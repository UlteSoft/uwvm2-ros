cmake_minimum_required(VERSION 3.20)
project(uwvm_nvidia_groot_export LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
 message(FATAL_ERROR "Explicit actual current Source checkout required")
endif()
include("${UWVM_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
if(NOT IS_ABSOLUTE "${UWVM_NVIDIA_BUNDLE}")
 message(FATAL_ERROR "Explicit actual generated whole bundle required")
endif()
foreach(_leaf native_identity.cc native_constants.hpp generation.vendor.json)
 if(NOT EXISTS "${UWVM_NVIDIA_BUNDLE}/${_leaf}")
  message(FATAL_ERROR "Actual vendor product missing: ${_leaf}")
 endif()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(rosidl_runtime_c REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
set(_targets rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
find_package(uwvm_groot_bridge REQUIRED)
if(NOT TARGET uwvm_groot_bridge::uwvm_groot_bridge__rosidl_generator_cpp)
 message(FATAL_ERROR "Real installed target missing: uwvm_groot_bridge::uwvm_groot_bridge__rosidl_generator_cpp")
endif()
list(APPEND _targets uwvm_groot_bridge::uwvm_groot_bridge__rosidl_generator_cpp)
if(NOT TARGET uwvm_groot_bridge::uwvm_groot_bridge__rosidl_typesupport_cpp)
 message(FATAL_ERROR "Real installed target missing: uwvm_groot_bridge::uwvm_groot_bridge__rosidl_typesupport_cpp")
endif()
list(APPEND _targets uwvm_groot_bridge::uwvm_groot_bridge__rosidl_typesupport_cpp)
uwvm_current_native_link_environment(${_targets})
add_executable(uwvm-nvidia-groot-export export_identities.cpp)
target_include_directories(uwvm-nvidia-groot-export PRIVATE "${UWVM_SOURCE_ROOT}" "${UWVM_NVIDIA_BUNDLE}")
target_compile_definitions(uwvm-nvidia-groot-export PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_link_libraries(uwvm-nvidia-groot-export PRIVATE ${_targets})
