cmake_minimum_required(VERSION 3.20)
project(uwvm_nvidia_isaac_ros_guest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
 message(FATAL_ERROR "Actual selected wasm32 WASI toolchain required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/sdk/include/roswasm/ros2.h" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include/fast_io.h")
 message(FATAL_ERROR "Explicit full current Source/SDK/vendored I/O closure required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_NVIDIA_BOUND_BUNDLE}" OR NOT EXISTS "${UWVM_NVIDIA_BOUND_BUNDLE}/bounded_records.hpp" OR NOT EXISTS "${UWVM_NVIDIA_BOUND_BUNDLE}/generation.vendor.json")
 message(FATAL_ERROR "Actual bound complete generated record bundle required")
endif()
option(UWVM_NVIDIA_POSITIVE "Explicit distinct positive artifact; never a Runtime grant" OFF)
add_executable(nvidia-isaac_ros-telemetry telemetry.cpp)
target_include_directories(nvidia-isaac_ros-telemetry PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include" "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include" "${UWVM_NVIDIA_BOUND_BUNDLE}")
target_compile_definitions(nvidia-isaac_ros-telemetry PRIVATE UWVM_NVIDIA_POSITIVE=$<BOOL:${UWVM_NVIDIA_POSITIVE}>)
add_executable(nvidia-raw-denied raw_denied.cpp)
target_include_directories(nvidia-raw-denied PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include")
