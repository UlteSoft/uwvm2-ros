cmake_minimum_required(VERSION 3.16)
project(uwvm_ros2_installed_source_witness LANGUAGES CXX)
option(UWVM_ROS2_BUILD_INSTALLED_SOURCE "Build explicitly selected native-source witnesses" OFF)
if(NOT UWVM_ROS2_BUILD_INSTALLED_SOURCE)
  return()
endif()
set(UWVM_SOURCE "" CACHE PATH "Admitted current complete source root")
set(UWVM_NATIVE_PACKAGE "" CACHE STRING "One explicitly installed native Message package")
set(UWVM_NATIVE_PACKAGES "" CACHE STRING "Explicit installed packages for one coherent merged vendor/protocol catalog")
set(UWVM_SOURCE_GENERATION "" CACHE PATH "Fresh original installed-source proof products")
set(UWVM_BOUND_GENERATION "" CACHE PATH "Optional same-domain genuinely bound Native products")
set(UWVM_NATIVE_CODEC_ORACLE "" CACHE PATH "Optional complete independent native codec oracle")
set(UWVM_FAST_IO_INCLUDE "${UWVM_SOURCE}/third-parties/fast_io/include" CACHE PATH "Actual reviewed fast_io include tree for the C++20 production Gateway")
option(UWVM_INSTALLED_PROTOCOL_PROOF "Build explicitly selected complete legacy Service/Action contracts" OFF)
set(UWVM_PROTOCOL_VALUE_ORACLE "" CACHE PATH "Optional matching independent complete native protocol value oracle")
if(NOT UWVM_NATIVE_CODEC_ORACLE STREQUAL "" AND UWVM_BOUND_GENERATION STREQUAL "")
  message(FATAL_ERROR "A native codec oracle requires its actual same-domain bound generation")
endif()
foreach(path IN ITEMS UWVM_SOURCE UWVM_SOURCE_GENERATION)
  if(NOT IS_ABSOLUTE "${${path}}" OR NOT IS_DIRECTORY "${${path}}")
    message(FATAL_ERROR "${path} requires an explicit physical admitted directory")
  endif()
endforeach()
set(selected_native_packages ${UWVM_NATIVE_PACKAGES})
if(NOT UWVM_NATIVE_PACKAGE STREQUAL "")
  list(APPEND selected_native_packages "${UWVM_NATIVE_PACKAGE}")
endif()
list(REMOVE_DUPLICATES selected_native_packages)
if(NOT selected_native_packages)
  message(FATAL_ERROR "Select the actual installed native packages for this one coherent catalog")
endif()
foreach(package IN LISTS selected_native_packages)
  if(NOT package MATCHES "^[a-z][a-z0-9_]*$")
    message(FATAL_ERROR "Every selected native package must be an exact ROS package identifier")
  endif()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
foreach(package IN LISTS selected_native_packages)
  find_package(${package} REQUIRED)
endforeach()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
  message(FATAL_ERROR "Actual full-MD/native link trace requires GNU or Clang")
endif()
function(admit_native_target target)
  target_include_directories(${target} PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/src" "${UWVM_SOURCE_GENERATION}")
  target_compile_definitions(${target} PRIVATE UWVM_ROS2_HAVE_OPENSSL=1)
  ament_target_dependencies(${target} rclcpp rosidl_typesupport_introspection_cpp ${selected_native_packages})
  target_link_libraries(${target} OpenSSL::Crypto)
  target_compile_options(${target} PRIVATE -O1 -g0 -UNDEBUG -fno-fast-math -ffp-contract=off -MD)
  target_link_options(${target} PRIVATE -Wl,--trace)
endfunction()
add_executable(native-source-identities "${UWVM_SOURCE_GENERATION}/native_source_identity.cc")
admit_native_target(native-source-identities)
add_executable(native-source-contract "${UWVM_SOURCE}/tests/ros2_compat/installed_source_contract.cc")
admit_native_target(native-source-contract)
if(NOT UWVM_BOUND_GENERATION STREQUAL "")
  if(NOT IS_ABSOLUTE "${UWVM_BOUND_GENERATION}" OR NOT EXISTS "${UWVM_BOUND_GENERATION}/native_ros2.hpp" OR
     NOT EXISTS "${UWVM_BOUND_GENERATION}/manifest.hpp")
    message(FATAL_ERROR "The original matching bound generation is required")
  endif()
  add_executable(native-source-binding "${UWVM_SOURCE}/tests/ros2_compat/installed_source_binding.cc")
  admit_native_target(native-source-binding)
  target_include_directories(native-source-binding PRIVATE "${UWVM_BOUND_GENERATION}")
  target_compile_definitions(native-source-binding PRIVATE
    UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp"
    UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_SOURCE_GENERATION}/native_source_provider.hpp")
  add_executable(native-source-unproved-binding "${UWVM_SOURCE}/tests/ros2_compat/installed_source_binding.cc")
  admit_native_target(native-source-unproved-binding)
  target_include_directories(native-source-unproved-binding PRIVATE "${UWVM_BOUND_GENERATION}")
  target_compile_definitions(native-source-unproved-binding PRIVATE
    UWVM_EXPECT_SOURCE_REFUSAL=1
    UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp"
    UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_BOUND_GENERATION}/native_ros2.hpp")
  if(NOT UWVM_NATIVE_CODEC_ORACLE STREQUAL "")
    if(NOT IS_ABSOLUTE "${UWVM_NATIVE_CODEC_ORACLE}" OR NOT EXISTS "${UWVM_NATIVE_CODEC_ORACLE}/contract.c")
      message(FATAL_ERROR "The original independently generated complete native oracle is required")
    endif()
    if(NOT IS_ABSOLUTE "${UWVM_FAST_IO_INCLUDE}" OR NOT EXISTS "${UWVM_FAST_IO_INCLUDE}/fast_io_freestanding.h")
      message(FATAL_ERROR "The C++20 production Gateway requires its actual admitted fast_io headers")
    endif()
    add_executable(native-source-codec-oracle "${UWVM_NATIVE_CODEC_ORACLE}/contract.c")
    set_source_files_properties("${UWVM_NATIVE_CODEC_ORACLE}/contract.c" PROPERTIES LANGUAGE CXX)
    admit_native_target(native-source-codec-oracle)
    set_target_properties(native-source-codec-oracle PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
    target_include_directories(native-source-codec-oracle PRIVATE "${UWVM_BOUND_GENERATION}" "${UWVM_SOURCE}/sdk/include" "${UWVM_FAST_IO_INCLUDE}")
    target_compile_definitions(native-source-codec-oracle PRIVATE
      UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp"
      UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_SOURCE_GENERATION}/native_source_provider.hpp")
  endif()
  if(UWVM_INSTALLED_PROTOCOL_PROOF)
    add_executable(native-protocol-contract "${UWVM_SOURCE}/tests/ros2_compat/installed_protocol_contract.cc")
    admit_native_target(native-protocol-contract)
    target_include_directories(native-protocol-contract PRIVATE "${UWVM_BOUND_GENERATION}")
    target_compile_definitions(native-protocol-contract PRIVATE
      UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp"
      UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_SOURCE_GENERATION}/native_source_provider.hpp")
    foreach(target IN ITEMS native-protocol-binding native-protocol-unproved-binding)
      add_executable(${target} "${UWVM_SOURCE}/tests/ros2_compat/installed_protocol_binding.cc")
      admit_native_target(${target})
      target_include_directories(${target} PRIVATE "${UWVM_BOUND_GENERATION}")
      target_compile_definitions(${target} PRIVATE UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp")
    endforeach()
    target_compile_definitions(native-protocol-binding PRIVATE
      UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_SOURCE_GENERATION}/native_source_provider.hpp")
    target_compile_definitions(native-protocol-unproved-binding PRIVATE UWVM_EXPECT_SOURCE_REFUSAL=1
      UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_BOUND_GENERATION}/native_ros2.hpp")
    if(NOT UWVM_PROTOCOL_VALUE_ORACLE STREQUAL "")
      if(NOT IS_ABSOLUTE "${UWVM_PROTOCOL_VALUE_ORACLE}" OR NOT EXISTS "${UWVM_PROTOCOL_VALUE_ORACLE}/contract.cc" OR
         NOT EXISTS "${UWVM_FAST_IO_INCLUDE}/fast_io_freestanding.h")
        message(FATAL_ERROR "The actual complete protocol value oracle and reviewed fast_io tree are required")
      endif()
      add_executable(native-protocol-values "${UWVM_PROTOCOL_VALUE_ORACLE}/contract.cc")
      admit_native_target(native-protocol-values)
      set_target_properties(native-protocol-values PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
      target_include_directories(native-protocol-values PRIVATE "${UWVM_BOUND_GENERATION}" "${UWVM_SOURCE}/sdk/include" "${UWVM_FAST_IO_INCLUDE}")
      target_compile_definitions(native-protocol-values PRIVATE
        UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_BOUND_GENERATION}/manifest.hpp"
        UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_SOURCE_GENERATION}/native_source_provider.hpp")
    endif()
  elseif(NOT UWVM_PROTOCOL_VALUE_ORACLE STREQUAL "")
    message(FATAL_ERROR "The protocol value oracle requires explicit complete protocol proof selection")
  endif()
endif()
