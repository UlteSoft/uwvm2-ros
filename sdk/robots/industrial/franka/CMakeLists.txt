cmake_minimum_required(VERSION 3.20)
project(uwvm_franka LANGUAGES C CXX)
# Every lane and variant requires explicit selection; source grants nothing.
option(UWVM_FRANKA_BUILD_CONTRACTS "Build authored whole-value SDK contracts" OFF)
option(UWVM_FRANKA_BUILD_WASI_EXAMPLES "Build genuine WASI C/C++ entries" OFF)
set(UWVM_FRANKA_VARIANT "" CACHE STRING "Exactly ros1 or ros2")
set_property(CACHE UWVM_FRANKA_VARIANT PROPERTY STRINGS ros1 ros2)
set(UWVM_FRANKA_SOURCE_ROOT "" CACHE PATH "Absolute actual source snapshot")
set(UWVM_FRANKA_GENERATED "" CACHE PATH "Absolute matching complete source bundle")
if(NOT UWVM_FRANKA_BUILD_CONTRACTS AND NOT UWVM_FRANKA_BUILD_WASI_EXAMPLES)
  return()
endif()
if(NOT UWVM_FRANKA_VARIANT MATCHES "^(ros1|ros2)$")
  message(FATAL_ERROR "Explicit selected original Franka variant required")
endif()
foreach(path IN ITEMS "${UWVM_FRANKA_SOURCE_ROOT}" "${UWVM_FRANKA_GENERATED}")
  if(NOT IS_ABSOLUTE "${path}" OR NOT IS_DIRECTORY "${path}")
    message(FATAL_ERROR "Actual absolute source and generated paths required")
  endif()
endforeach()
file(READ "${UWVM_FRANKA_GENERATED}/generation.vendor.json" generation)
file(READ "${UWVM_FRANKA_GENERATED}/preparation.vendor.json" preparation)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/catalogs/${UWVM_FRANKA_VARIANT}/catalog.json" original_sha)
string(JSON captured_sha GET "${preparation}" original_catalog sha256)
if(NOT captured_sha STREQUAL original_sha)
  message(FATAL_ERROR "Selected original Franka family differs from supplied bundle")
endif()
string(JSON count LENGTH "${generation}" outputs)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON name MEMBER "${generation}" outputs ${index})
  if(name MATCHES "(^/|(^|/)[.][.](/|$))" OR IS_SYMLINK "${UWVM_FRANKA_GENERATED}/${name}")
    message(FATAL_ERROR "Generated path must remain inside exact bundle")
  endif()
  string(JSON expected_sha GET "${generation}" outputs "${name}" sha256)
  string(JSON expected_bytes GET "${generation}" outputs "${name}" bytes)
  file(SHA256 "${UWVM_FRANKA_GENERATED}/${name}" actual_sha)
  file(SIZE "${UWVM_FRANKA_GENERATED}/${name}" actual_bytes)
  if(NOT expected_sha STREQUAL actual_sha OR NOT expected_bytes EQUAL actual_bytes)
    message(FATAL_ERROR "Exact generated product differs: ${name}")
  endif()
endforeach()
string(SUBSTRING "${UWVM_FRANKA_VARIANT}" 3 1 variant_version)
add_library(franka_selected INTERFACE)
target_include_directories(franka_selected INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${UWVM_FRANKA_GENERATED}" "${UWVM_FRANKA_SOURCE_ROOT}/sdk/include")
target_compile_definitions(franka_selected INTERFACE "UWVM_FRANKA_ROS_VERSION=${variant_version}")
if(UWVM_FRANKA_BUILD_CONTRACTS)
  add_executable(franka_contract_c "tests/contract_${UWVM_FRANKA_VARIANT}.c")
  add_executable(franka_contract_cpp "tests/contract_${UWVM_FRANKA_VARIANT}.cc")
  target_compile_options(franka_contract_c PRIVATE -UNDEBUG)
  target_compile_options(franka_contract_cpp PRIVATE -UNDEBUG)
  set_property(TARGET franka_contract_c PROPERTY C_STANDARD 11)
  set_property(TARGET franka_contract_cpp PROPERTY CXX_STANDARD 20)
  target_link_libraries(franka_contract_c PRIVATE franka_selected)
  target_link_libraries(franka_contract_cpp PRIVATE franka_selected)
endif()
if(UWVM_FRANKA_BUILD_WASI_EXAMPLES)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
    message(FATAL_ERROR "Actual WASI toolchain required for Guest commands")
  endif()
  add_executable(franka_guest_c examples/guest.c)
  add_executable(franka_guest_cpp examples/guest.cc)
  set_property(TARGET franka_guest_c PROPERTY C_STANDARD 11)
  set_property(TARGET franka_guest_cpp PROPERTY CXX_STANDARD 20)
  target_link_libraries(franka_guest_c PRIVATE franka_selected)
  target_link_libraries(franka_guest_cpp PRIVATE franka_selected)
endif()
foreach(target IN ITEMS franka_contract_c franka_contract_cpp franka_guest_c franka_guest_cpp)
  if(TARGET ${target} AND CMAKE_SYSTEM_NAME STREQUAL "WASI")
    set_target_properties(${target} PROPERTIES SUFFIX ".wasm")
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>")
    target_link_options(${target} PRIVATE "-Wl,--threads=1")
  endif()
endforeach()
