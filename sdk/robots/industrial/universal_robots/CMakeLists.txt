cmake_minimum_required(VERSION 3.20)
project(uwvm_universal_robots LANGUAGES C CXX)
# Every lane and variant requires explicit selection; source grants nothing.
option(UWVM_UR_BUILD_CONTRACTS "Build authored whole-value SDK contracts" OFF)
option(UWVM_UR_BUILD_WASI_EXAMPLES "Build genuine WASI C/C++ entries" OFF)
set(UWVM_UR_VARIANT "" CACHE STRING "Exactly ros1, ros2 or polyscopex")
set_property(CACHE UWVM_UR_VARIANT PROPERTY STRINGS ros1 ros2 polyscopex)
set(UWVM_UR_SOURCE_ROOT "" CACHE PATH "Absolute actual source snapshot")
set(UWVM_UR_GENERATED "" CACHE PATH "Absolute matching complete source bundle")
if(NOT UWVM_UR_BUILD_CONTRACTS AND NOT UWVM_UR_BUILD_WASI_EXAMPLES)
  return()
endif()
if(NOT UWVM_UR_VARIANT MATCHES "^(ros1|ros2|polyscopex)$")
  message(FATAL_ERROR "Explicit selected original UR variant required")
endif()
foreach(path IN ITEMS "${UWVM_UR_SOURCE_ROOT}" "${UWVM_UR_GENERATED}")
  if(NOT IS_ABSOLUTE "${path}" OR NOT IS_DIRECTORY "${path}")
    message(FATAL_ERROR "Actual absolute source and generated paths required")
  endif()
endforeach()
file(READ "${UWVM_UR_GENERATED}/generation.vendor.json" generation)
file(READ "${UWVM_UR_GENERATED}/preparation.vendor.json" preparation)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/catalogs/${UWVM_UR_VARIANT}/catalog.json" original_sha)
string(JSON captured_sha GET "${preparation}" original_catalog sha256)
if(NOT captured_sha STREQUAL original_sha)
  message(FATAL_ERROR "Selected original UR family differs from supplied bundle")
endif()
string(JSON count LENGTH "${generation}" outputs)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON name MEMBER "${generation}" outputs ${index})
  if(name MATCHES "(^/|(^|/)[.][.](/|$))" OR IS_SYMLINK "${UWVM_UR_GENERATED}/${name}")
    message(FATAL_ERROR "Generated path must remain inside exact bundle")
  endif()
  string(JSON expected_sha GET "${generation}" outputs "${name}" sha256)
  string(JSON expected_bytes GET "${generation}" outputs "${name}" bytes)
  file(SHA256 "${UWVM_UR_GENERATED}/${name}" actual_sha)
  file(SIZE "${UWVM_UR_GENERATED}/${name}" actual_bytes)
  if(NOT expected_sha STREQUAL actual_sha OR NOT expected_bytes EQUAL actual_bytes)
    message(FATAL_ERROR "Exact generated product differs: ${name}")
  endif()
endforeach()
string(TOUPPER "${UWVM_UR_VARIANT}" variant_macro)
add_library(ur_selected INTERFACE)
target_include_directories(ur_selected INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${UWVM_UR_GENERATED}" "${UWVM_UR_SOURCE_ROOT}/sdk/include")
target_compile_definitions(ur_selected INTERFACE "UWVM_UR_${variant_macro}=1")
if(UWVM_UR_BUILD_CONTRACTS)
  add_executable(ur_contract_c "tests/contract_${UWVM_UR_VARIANT}.c")
  add_executable(ur_contract_cpp "tests/contract_${UWVM_UR_VARIANT}.cc")
  target_compile_options(ur_contract_c PRIVATE -UNDEBUG)
  target_compile_options(ur_contract_cpp PRIVATE -UNDEBUG)
  set_property(TARGET ur_contract_c PROPERTY C_STANDARD 11)
  set_property(TARGET ur_contract_cpp PROPERTY CXX_STANDARD 20)
  target_link_libraries(ur_contract_c PRIVATE ur_selected)
  target_link_libraries(ur_contract_cpp PRIVATE ur_selected)
endif()
if(UWVM_UR_BUILD_WASI_EXAMPLES)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
    message(FATAL_ERROR "Actual WASI toolchain required for Guest commands")
  endif()
  add_executable(ur_guest_c examples/guest.c)
  add_executable(ur_guest_cpp examples/guest.cc)
  set_property(TARGET ur_guest_c PROPERTY C_STANDARD 11)
  set_property(TARGET ur_guest_cpp PROPERTY CXX_STANDARD 20)
  target_link_libraries(ur_guest_c PRIVATE ur_selected)
  target_link_libraries(ur_guest_cpp PRIVATE ur_selected)
endif()
foreach(target IN ITEMS ur_contract_c ur_contract_cpp ur_guest_c ur_guest_cpp)
  if(TARGET ${target} AND CMAKE_SYSTEM_NAME STREQUAL "WASI")
    set_target_properties(${target} PROPERTIES SUFFIX ".wasm")
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>")
    target_link_options(${target} PRIVATE "-Wl,--threads=1")
  endif()
endforeach()
