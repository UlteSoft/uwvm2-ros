cmake_minimum_required(VERSION 3.20)
project(uwvm_industrial_examples LANGUAGES C CXX)
set(UWVM_SOURCE "" CACHE PATH "Complete reviewed SDK source checkout")
set(UWVM_INDUSTRIAL_GENERATION "" CACHE PATH "One actual same-source generated complete-v1 output")
set(UWVM_INDUSTRIAL_VENDOR "" CACHE STRING "fanuc, kuka, yaskawa or abb")
set(UWVM_INDUSTRIAL_ROS_VERSION 2 CACHE STRING "Explicit1/2; ROS1 example is ABB query only")
option(UWVM_INDUSTRIAL_BUILD_GUEST "Build actual Wasm source products" OFF)
option(UWVM_INDUSTRIAL_BUILD_NATIVE_FIXTURE "Build genuine ROS2 synthetic opposite process" OFF)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/sdk/include/roswasm/roswasmcpp.hpp")
 message(FATAL_ERROR "An actual complete SDK source tree is required")
endif()
if(NOT UWVM_INDUSTRIAL_VENDOR MATCHES "^(fanuc|kuka|yaskawa|abb)$")
 message(FATAL_ERROR "Select one complete vendor catalog/build domain explicitly")
endif()
if(NOT UWVM_INDUSTRIAL_ROS_VERSION MATCHES "^[12]$")
 message(FATAL_ERROR "Select ROS major1 or2")
endif()
if(NOT IS_ABSOLUTE "${UWVM_INDUSTRIAL_GENERATION}" OR NOT EXISTS "${UWVM_INDUSTRIAL_GENERATION}/generation.vendor.json")
 message(FATAL_ERROR "Actual reviewed generated Source revision is required; this CMake does not generate it")
endif()
foreach(_leaf bounded_records.h bounded_records.hpp bounded_interfaces.h bounded_interfaces.hpp)
 if(NOT EXISTS "${UWVM_INDUSTRIAL_GENERATION}/${_leaf}")
  message(FATAL_ERROR "Missing actual complete generated product ${_leaf}")
 endif()
endforeach()
set(_vendor "${CMAKE_CURRENT_LIST_DIR}/../${UWVM_INDUSTRIAL_VENDOR}")
if(UWVM_INDUSTRIAL_BUILD_GUEST)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  message(FATAL_ERROR "Guest examples require an actual WASI toolchain, not a host compiler")
 endif()
 if(UWVM_INDUSTRIAL_ROS_VERSION STREQUAL "2")
  add_executable(industrial_trajectory "${_vendor}/examples/trajectory.cpp")
  target_compile_features(industrial_trajectory PRIVATE cxx_std_20)
  target_include_directories(industrial_trajectory PRIVATE "${UWVM_SOURCE}/sdk/include" "${UWVM_INDUSTRIAL_GENERATION}")
  target_compile_definitions(industrial_trajectory PRIVATE UWVM_INDUSTRIAL_GENERATED_RECORDS="bounded_records.hpp" UWVM_INDUSTRIAL_GENERATED_INTERFACES="bounded_interfaces.hpp")
  target_compile_options(industrial_trajectory PRIVATE -fno-fast-math -ffp-contract=off)
  # The portable SDK uses numeric ROS errors. This WASI profile needs no Wasm
  # exception proposal; allocation exhaustion follows libc++ termination.
  target_compile_options(industrial_trajectory PRIVATE -fno-exceptions -fno-rtti)
  target_link_options(industrial_trajectory PRIVATE -nostartfiles -Wl,--no-entry -Wl,--export-memory)
  if(UWVM_INDUSTRIAL_VENDOR STREQUAL "kuka")
   set(_c_source "${_vendor}/examples/status.c")
  else()
   set(_c_source "${_vendor}/examples/query.c")
  endif()
 elseif(UWVM_INDUSTRIAL_VENDOR STREQUAL "abb")
  set(_c_source "${_vendor}/examples/query_ros1.c")
 else()
  message(FATAL_ERROR "No invented ROS1 backend for this selected vendor")
 endif()
 add_executable(industrial_c "${_c_source}")
 target_compile_features(industrial_c PRIVATE c_std_11)
 target_include_directories(industrial_c PRIVATE "${UWVM_SOURCE}/sdk/include" "${UWVM_INDUSTRIAL_GENERATION}")
 target_compile_options(industrial_c PRIVATE -fno-fast-math -ffp-contract=off)
 target_link_options(industrial_c PRIVATE -nostartfiles -Wl,--no-entry -Wl,--export-memory)
 foreach(_guest IN ITEMS industrial_c industrial_trajectory)
  if(TARGET ${_guest})
   set_target_properties(${_guest} PROPERTIES SUFFIX ".wasm")
   # Full bounded trajectory values exceed a linker's common 64KiB default.
   # Keep both reservation and ceiling explicit and the linker serial.
   target_link_options(${_guest} PRIVATE -Wl,--threads=1 -Wl,-z,stack-size=1048576
     -Wl,--initial-memory=2097152 -Wl,--max-memory=33554432)
  endif()
 endforeach()
endif()
if(UWVM_INDUSTRIAL_BUILD_NATIVE_FIXTURE)
 if(CMAKE_SYSTEM_NAME STREQUAL "WASI" OR NOT UWVM_INDUSTRIAL_ROS_VERSION STREQUAL "2")
  message(FATAL_ERROR "Native fixture is a separate real modern ROS2 host build")
 endif()
 foreach(_leaf manifest.hpp native_ros2.hpp native_constants.hpp)
  if(NOT EXISTS "${UWVM_INDUSTRIAL_GENERATION}/${_leaf}")
   message(FATAL_ERROR "A bound actual native Source revision is required")
  endif()
 endforeach()
 find_package(ament_cmake REQUIRED)
 find_package(rclcpp REQUIRED)
 find_package(rclcpp_action REQUIRED)
 find_package(control_msgs REQUIRED)
 find_package(sensor_msgs REQUIRED)
 set(UWVM_FAST_IO_INCLUDE "" CACHE PATH "Actual reviewed fast_io include root")
 if(NOT EXISTS "${UWVM_FAST_IO_INCLUDE}/fast_io.h")
  message(FATAL_ERROR "Actual fast_io C++20 source is required, no fake shim")
 endif()
 add_executable(industrial_native_fixture native_fixture_ros2.cc)
 target_compile_features(industrial_native_fixture PRIVATE cxx_std_20)
 target_include_directories(industrial_native_fixture PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/src" "${UWVM_INDUSTRIAL_GENERATION}" "${UWVM_FAST_IO_INCLUDE}")
 target_compile_definitions(industrial_native_fixture PRIVATE UWVM_INDUSTRIAL_NATIVE_HEADER="native_ros2.hpp" UWVM_ROS_CUSTOM_TOPIC_MANIFEST="manifest.hpp")
 ament_target_dependencies(industrial_native_fixture rclcpp rclcpp_action control_msgs sensor_msgs)
 # All additional selected native package/type-support usage requirements must
 # be supplied explicitly by the coherent complete-catalog SDK build. Merely
 # finding control_msgs does not close FANUC/KUKA/Yaskawa/ABB generated headers.
 set(UWVM_INDUSTRIAL_NATIVE_PACKAGES "" CACHE STRING "Every actual generated ROS package in this catalog; explicit semicolon list")
 if(NOT UWVM_INDUSTRIAL_NATIVE_PACKAGES)
  message(FATAL_ERROR "Declare every selected package; CMake does not infer or download a native SDK")
 endif()
 foreach(_package IN LISTS UWVM_INDUSTRIAL_NATIVE_PACKAGES)
  if(NOT _package MATCHES "^[a-z][a-z0-9_]*$")
   message(FATAL_ERROR "Actual ROS package name required: ${_package}")
  endif()
  find_package(${_package} CONFIG REQUIRED)
 endforeach()
 set(UWVM_INDUSTRIAL_NATIVE_PACKAGE_TARGETS "" CACHE STRING "Actual imported targets for EVERY selected package in this one native catalog")
 if(NOT UWVM_INDUSTRIAL_NATIVE_PACKAGE_TARGETS)
  message(FATAL_ERROR "Complete actual generated package target closure is required")
 endif()
 foreach(_target IN LISTS UWVM_INDUSTRIAL_NATIVE_PACKAGE_TARGETS)
  if(NOT TARGET "${_target}")
   message(FATAL_ERROR "Actual imported package target not found: ${_target}")
  endif()
 endforeach()
 target_link_libraries(industrial_native_fixture PRIVATE ${UWVM_INDUSTRIAL_NATIVE_PACKAGE_TARGETS})
endif()
# No add_test, try_run, source generation, install, ROS process or robot is run.
