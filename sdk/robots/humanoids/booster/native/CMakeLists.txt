cmake_minimum_required(VERSION 3.20)
project(uwvm_booster_native_contracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/tests/ros_custom_types/bounded-v2/native_protocol.h")
  message(FATAL_ERROR "Explicit current whole Source checkout required")
endif()
include("${UWVM_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
# Actual generated products are external inputs, not checked-in fake Native
# headers. Their source proof/full-MD/link/export must be independently retained.
foreach(_bundle UWVM_BOOSTER_ORIGINAL_BUNDLE UWVM_BOOSTER_BRIDGE_BUNDLE)
 if(NOT IS_ABSOLUTE "${${_bundle}}")
  message(FATAL_ERROR "Explicit actual generated bundle required: ${_bundle}")
 endif()
 foreach(_leaf native_identity.cc native_constants.hpp generation.vendor.json)
  if(NOT EXISTS "${${_bundle}}/${_leaf}")
   message(FATAL_ERROR "Real complete vendor product missing: ${_bundle}/${_leaf}")
  endif()
 endforeach()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Threads REQUIRED)
find_package(rosidl_runtime_c REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
foreach(_base rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
 if(NOT TARGET "${_base}")
  message(FATAL_ERROR "Actual modern native base target required: ${_base}")
 endif()
endforeach()
find_package(booster_interface REQUIRED)
find_package(uwvm_booster_sdk_bridge REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
set(_original_targets rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
set(_bridge_targets rclcpp::rclcpp rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
foreach(_package booster_interface std_msgs builtin_interfaces uwvm_booster_sdk_bridge)
  foreach(_provider rosidl_generator_cpp rosidl_typesupport_cpp)
    set(_target "${_package}::${_package}__${_provider}")
    if(NOT TARGET "${_target}")
      message(FATAL_ERROR "Actual full generated installed target required: ${_target}")
    endif()
    if(_package STREQUAL "uwvm_booster_sdk_bridge")
      list(APPEND _bridge_targets "${_target}")
    else()
      list(APPEND _original_targets "${_target}")
    endif()
  endforeach()
endforeach()
# The official ROS package and official SDK define DIFFERENT types in the same
# booster_interface namespace. They must never enter the same translation unit.
if(NOT IS_ABSOLUTE "${UWVM_BOOSTER_SDK_ROOT}" OR NOT EXISTS "${UWVM_BOOSTER_SDK_ROOT}/include/booster/robot/channel/channel_factory.hpp")
  message(FATAL_ERROR "Explicit actual fixed official SDK include closure required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_BOOSTER_LIBRARY}" OR NOT EXISTS "${UWVM_BOOSTER_LIBRARY}")
  message(FATAL_ERROR "Explicit real SDK static library required; no guessed search path")
endif()
string(LENGTH "${UWVM_BOOSTER_LIBRARY_SHA256}" _length)
if(NOT _length EQUAL 64 OR NOT UWVM_BOOSTER_LIBRARY_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Actual selected SDK library SHA256 required")
endif()
file(SHA256 "${UWVM_BOOSTER_LIBRARY}" _actual)
if(NOT _actual STREQUAL UWVM_BOOSTER_LIBRARY_SHA256)
  message(FATAL_ERROR "Native SDK library selection differs")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The selected vendor library and finite native bridge require Linux")
endif()
add_library(booster-sdk-selected-current STATIC IMPORTED)
set_target_properties(booster-sdk-selected-current PROPERTIES IMPORTED_LOCATION "${UWVM_BOOSTER_LIBRARY}"
 INTERFACE_INCLUDE_DIRECTORIES "${UWVM_BOOSTER_SDK_ROOT}/include")
set(_sdk_targets ${_bridge_targets} booster-sdk-selected-current Threads::Threads ${CMAKE_DL_LIBS} rt)
uwvm_current_native_link_mode_guard("${CMAKE_DL_LIBS};rt")
uwvm_current_native_link_environment(${_original_targets} ${_bridge_targets} booster-sdk-selected-current Threads::Threads)
add_executable(booster-export-identities export_identities.cpp)
target_include_directories(booster-export-identities PRIVATE "${UWVM_SOURCE_ROOT}" "${UWVM_BOOSTER_ORIGINAL_BUNDLE}")
target_compile_definitions(booster-export-identities PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_link_libraries(booster-export-identities PRIVATE ${_original_targets})
add_executable(booster-sdk-export-identities export_sdk_bridge_identities.cpp)
target_include_directories(booster-sdk-export-identities PRIVATE "${UWVM_SOURCE_ROOT}" "${UWVM_BOOSTER_BRIDGE_BUNDLE}")
target_compile_definitions(booster-sdk-export-identities PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_link_libraries(booster-sdk-export-identities PRIVATE ${_bridge_targets})
option(UWVM_BOOSTER_ENABLE_NATIVE_SEND "Explicit separately owned finite T1 SDK process" OFF)
add_executable(booster-sdk-native-bridge sdk_bridge_main.cpp)
target_include_directories(booster-sdk-native-bridge PRIVATE "${UWVM_SOURCE_ROOT}")
target_compile_definitions(booster-sdk-native-bridge PRIVATE UWVM_BOOSTER_SDK_BRIDGE_ENABLE=$<BOOL:${UWVM_BOOSTER_ENABLE_NATIVE_SEND}>)
target_link_libraries(booster-sdk-native-bridge PRIVATE ${_sdk_targets})
add_executable(booster-sdk-value-contract sdk_value_contract.cpp)
target_link_libraries(booster-sdk-value-contract PRIVATE ${_sdk_targets})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(booster-sdk-value-contract PRIVATE -UNDEBUG)
elseif(MSVC)
 target_compile_options(booster-sdk-value-contract PRIVATE /UNDEBUG)
endif()
# Executable links instantiate real TypeSupport / complete values. They are
# future qualification steps, not a precomputed native-compatibility boolean.

option(UWVM_BOOSTER_ENABLE_NATIVE_OBSERVE "Explicit read-only finite model/firmware/mode query process" OFF)
add_executable(booster-sdk-observe sdk_observe.cpp)
target_include_directories(booster-sdk-observe PRIVATE "${UWVM_SOURCE_ROOT}")
target_compile_definitions(booster-sdk-observe PRIVATE UWVM_BOOSTER_SDK_OBSERVE_ENABLE=$<BOOL:${UWVM_BOOSTER_ENABLE_NATIVE_OBSERVE}>)
target_link_libraries(booster-sdk-observe PRIVATE booster-sdk-selected-current Threads::Threads ${CMAKE_DL_LIBS} rt)
