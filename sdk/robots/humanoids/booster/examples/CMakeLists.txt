cmake_minimum_required(VERSION 3.20)
project(uwvm_booster_guest_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/sdk/include/roswasm/ros2.h")
 message(FATAL_ERROR "Explicit complete actual Source root required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_GENERATED_BUNDLE}" OR NOT EXISTS "${UWVM_GENERATED_BUNDLE}/bounded_records.hpp")
 message(FATAL_ERROR "Actual bound complete vendor output required; no fabricated header")
endif()
if(NOT CMAKE_CXX_COMPILER_TARGET MATCHES "^wasm32")
 message(FATAL_ERROR "Select the genuine WASI wasm32 compiler/toolchain; native ABI linking is forbidden")
endif()
option(UWVM_HUMANOID_POSITIVE "Select separate positive executable; grants remain external" OFF)
foreach(_name raw_command complete_command telemetry)
 add_executable(booster-${_name} "${_name}.cpp")
 target_include_directories(booster-${_name} PRIVATE "${UWVM_SOURCE_ROOT}/sdk/include"
  "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include" "${UWVM_GENERATED_BUNDLE}")
 target_compile_definitions(booster-${_name} PRIVATE UWVM_HUMANOID_POSITIVE=$<BOOL:${UWVM_HUMANOID_POSITIVE}>)
 # raw ROS imports are precisely linked by the real Wasm import mechanism.
 # No -allow-undefined blanket, native SDK, socket or management function.
endforeach()
