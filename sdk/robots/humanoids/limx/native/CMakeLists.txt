cmake_minimum_required(VERSION 3.20)
project(uwvm_limx_native_contracts LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Actual selected Linux SDK/process lane required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/tests/ros_custom_types/bounded-v2/native_protocol.h")
  message(FATAL_ERROR "Explicit current whole Source checkout required")
endif()
include("${UWVM_SOURCE_ROOT}/sdk/custom-types/native_current/SelectedNative.cmake")
# Actual generated products are external inputs, not checked-in fake Native
# headers. Their source proof/full-MD/link/export must be independently retained.
foreach(_bundle UWVM_LIMX_BRIDGE_BUNDLE)
 if(NOT IS_ABSOLUTE "${${_bundle}}")
  message(FATAL_ERROR "Explicit actual generated bundle required: ${_bundle}")
 endif()
 foreach(_leaf native_identity.cc native_constants.hpp generation.vendor.json)
  if(NOT EXISTS "${${_bundle}}/${_leaf}")
   message(FATAL_ERROR "Real complete vendor product missing: ${_bundle}/${_leaf}")
  endif()
 endforeach()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_runtime_c REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
foreach(_base rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
 if(NOT TARGET "${_base}")
  message(FATAL_ERROR "Actual modern native base target required: ${_base}")
 endif()
endforeach()
find_package(uwvm_limx_bridge REQUIRED)
set(_targets rclcpp::rclcpp rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
foreach(_provider rosidl_generator_cpp rosidl_typesupport_cpp)
  set(_target "uwvm_limx_bridge::uwvm_limx_bridge__${_provider}")
  if(NOT TARGET "${_target}")
    message(FATAL_ERROR "Build/install the actual full NEW bridge interface package first")
  endif()
  list(APPEND _targets "${_target}")
endforeach()
if(NOT IS_ABSOLUTE "${UWVM_LIMX_SDK_ROOT}" OR NOT EXISTS "${UWVM_LIMX_SDK_ROOT}/include/limxsdk/topic_channel.h")
  message(FATAL_ERROR "Actual pinned official SDK include closure required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_LIMX_LIBRARY}" OR NOT EXISTS "${UWVM_LIMX_LIBRARY}")
  message(FATAL_ERROR "Explicit actual installed native SDK library required")
endif()
string(LENGTH "${UWVM_LIMX_LIBRARY_SHA256}" _length)
if(NOT _length EQUAL 64 OR NOT UWVM_LIMX_LIBRARY_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Explicit native library selection hash required")
endif()
file(SHA256 "${UWVM_LIMX_LIBRARY}" _actual)
if(NOT _actual STREQUAL UWVM_LIMX_LIBRARY_SHA256)
  message(FATAL_ERROR "Native SDK library differs from actual selected bytes")
endif()
add_library(limx-selected-current SHARED IMPORTED)
set_target_properties(limx-selected-current PROPERTIES IMPORTED_LOCATION "${UWVM_LIMX_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${UWVM_LIMX_SDK_ROOT}/include")
list(APPEND _targets limx-selected-current)
uwvm_current_native_link_environment(${_targets})
add_executable(limx-export-identities export_identities.cpp)
target_include_directories(limx-export-identities PRIVATE "${UWVM_SOURCE_ROOT}" "${UWVM_LIMX_BRIDGE_BUNDLE}")
target_compile_definitions(limx-export-identities PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_link_libraries(limx-export-identities PRIVATE ${_targets})
option(UWVM_LIMX_ENABLE_NATIVE_SEND "Explicit finite separately owned native command process" OFF)
add_executable(limx-native-bridge bridge_main.cpp)
target_include_directories(limx-native-bridge PRIVATE "${UWVM_SOURCE_ROOT}")
target_compile_definitions(limx-native-bridge PRIVATE UWVM_LIMX_BRIDGE_ENABLE=$<BOOL:${UWVM_LIMX_ENABLE_NATIVE_SEND}>)
target_link_libraries(limx-native-bridge PRIVATE ${_targets})
add_executable(limx-value-contract value_contract.cpp)
target_link_libraries(limx-value-contract PRIVATE ${_targets})
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(limx-value-contract PRIVATE -UNDEBUG)
elseif(MSVC)
  target_compile_options(limx-value-contract PRIVATE /UNDEBUG)
endif()
