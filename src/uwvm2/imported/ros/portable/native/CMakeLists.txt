cmake_minimum_required(VERSION 3.22)
project(uwvm_portable_native_joint LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(rmw REQUIRED)
get_filename_component(UWVM_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_executable(uwvm-portable-native-joint ros2_joint_main.cc)
target_compile_features(uwvm-portable-native-joint PRIVATE cxx_std_23)
target_include_directories(uwvm-portable-native-joint PRIVATE "${UWVM_SOURCE_ROOT}/src" "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include")
target_compile_options(uwvm-portable-native-joint PRIVATE -Wall -Wextra -Werror -fexceptions -fasynchronous-unwind-tables)
ament_target_dependencies(uwvm-portable-native-joint rclcpp rclcpp_action control_msgs controller_manager_msgs sensor_msgs trajectory_msgs rmw)
