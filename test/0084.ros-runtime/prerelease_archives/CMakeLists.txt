cmake_minimum_required(VERSION 3.16)
project(prerelease_chatter_witness LANGUAGES CXX)
find_package(rclcpp REQUIRED)
find_package(rmw REQUIRED)
find_package(std_msgs REQUIRED)

target_link_options(jazzy_chatter PRIVATE "LINKER:--as-needed")
target_link_libraries(jazzy_chatter PRIVATE rclcpp::rclcpp rmw::rmw std_msgs::std_msgs__rosidl_typesupport_cpp)
