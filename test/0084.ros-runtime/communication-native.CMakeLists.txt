cmake_minimum_required(VERSION 3.22)
project(scene_communication_native LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rmw REQUIRED)
find_package(fastrtps REQUIRED)
set(UWVM_SCENE_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Frozen VM source root used for native qualification")
add_executable(communication_native communication_native_ros2.cpp)
target_compile_features(communication_native PRIVATE cxx_std_23)
target_compile_options(communication_native PRIVATE -Wall -Wextra -Werror -g -O1)
target_include_directories(communication_native PRIVATE "${UWVM_SCENE_SOURCE_ROOT}/src" "${UWVM_SCENE_SOURCE_ROOT}/third-parties/fast_io/include")
ament_target_dependencies(communication_native rclcpp std_msgs std_srvs rmw fastrtps)
