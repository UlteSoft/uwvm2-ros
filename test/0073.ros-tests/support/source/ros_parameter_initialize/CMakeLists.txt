cmake_minimum_required(VERSION 3.16)
project(uwvm_ros_parameter_initialize LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_executable(parameter_initialize_native native_ros2.cc)
target_compile_features(parameter_initialize_native PRIVATE cxx_std_20)
target_compile_options(parameter_initialize_native PRIVATE -UNDEBUG)
target_include_directories(parameter_initialize_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../src ${CMAKE_CURRENT_SOURCE_DIR}/../../third-parties/fast_io/include)
ament_target_dependencies(parameter_initialize_native rclcpp rcl_interfaces)
