cmake_minimum_required(VERSION 3.16)
project(uwvm_rclcpp_compat_real_native_peer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
set(UWVM_SOURCE "" CACHE PATH "Actual complete admitted source")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}")
  message(FATAL_ERROR "Actual source root required")
endif()
add_executable(rclcpp-compat-native-peer peer.cc)
# Native include paths deliberately do NOT include the Guest compatibility map.
target_include_directories(rclcpp-compat-native-peer PRIVATE "${UWVM_SOURCE}")
target_compile_features(rclcpp-compat-native-peer PRIVATE cxx_std_17)
target_compile_options(rclcpp-compat-native-peer PRIVATE -O1 -g0 -UNDEBUG -MD -Wno-unused-parameter)
target_link_options(rclcpp-compat-native-peer PRIVATE -Wl,-t)
ament_target_dependencies(rclcpp-compat-native-peer rclcpp std_msgs)
