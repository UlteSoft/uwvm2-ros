cmake_minimum_required(VERSION 3.16)
project(roswasm_rclcpp_service_native_peer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(example_interfaces REQUIRED)
add_executable(service_peer peer.cc)
target_compile_features(service_peer PRIVATE cxx_std_17)
ament_target_dependencies(service_peer rclcpp example_interfaces)
install(TARGETS service_peer DESTINATION lib/${PROJECT_NAME})
# Root must provide its actual coherent source and a fresh actual bound bundle.
set(UWVM_SOURCE_ROOT "" CACHE PATH "Actual whole-source root")
set(UWVM_BOUND_GENERATED "" CACHE PATH "Actual bound AddTwoInts generated directory")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}")
  message(FATAL_ERROR "Actual absolute whole-source path required")
endif()
add_executable(identity_export identity_export.cc)
target_compile_features(identity_export PRIVATE cxx_std_17)
target_compile_definitions(identity_export PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_include_directories(identity_export PRIVATE "${UWVM_SOURCE_ROOT}")
ament_target_dependencies(identity_export example_interfaces rclcpp)
option(UWVM_BUILD_BOUND_CONTRACT "Build only after actual native metadata bind+generation" OFF)
if(UWVM_BUILD_BOUND_CONTRACT)
  # Actual generation is a prior explicit owned stage, never run by configure.
  set(UWVM_BOUND_CATALOG "" CACHE FILEPATH "Actual same-source native-bound catalog")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  include("${UWVM_SOURCE_ROOT}/sdk/origin/custom-types/native_current/SelectedNative.cmake")
  add_executable(codec_contract codec_contract.cc)
  uwvm_select_current_native_contract(codec_contract
    "${UWVM_SOURCE_ROOT}" "${UWVM_BOUND_CATALOG}" "${UWVM_BOUND_GENERATED}"
    "${Python3_EXECUTABLE}" "example_interfaces/srv/AddTwoInts")
endif()
ament_package()
