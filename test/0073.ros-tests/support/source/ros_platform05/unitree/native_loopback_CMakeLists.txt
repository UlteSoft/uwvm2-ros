cmake_minimum_required(VERSION 3.20)
project(uwvm_unitree_finite_native_loopback LANGUAGES C CXX)
option(UWVM_UNITREE_NATIVE_LOOPBACK "Build the actual generated ROS2 API loopback peer" OFF)
if(NOT UWVM_UNITREE_NATIVE_LOOPBACK)
  return()
endif()
set(UWVM_UNITREE_SOURCE "" CACHE PATH "Absolute complete frozen source checkout")
set(UWVM_UNITREE_API_GENERATED "" CACHE PATH "Actual complete API8 generated family")
set(UWVM_UNITREE_PUBLIC_INCLUDE "" CACHE PATH "Absolute canonical installed SDK include root")
set(UWVM_UNITREE_FAST_IO "" CACHE PATH "Absolute matching canonical C++ foundation include root")
foreach(path IN ITEMS "${UWVM_UNITREE_SOURCE}" "${UWVM_UNITREE_API_GENERATED}")
  if(NOT IS_ABSOLUTE "${path}" OR NOT IS_DIRECTORY "${path}")
    message(FATAL_ERROR "Explicit complete source and generated API family paths required")
  endif()
endforeach()
if(NOT UWVM_UNITREE_PUBLIC_INCLUDE)
  set(UWVM_UNITREE_PUBLIC_INCLUDE "${UWVM_UNITREE_SOURCE}/sdk/include")
endif()
if(NOT UWVM_UNITREE_FAST_IO)
  set(UWVM_UNITREE_FAST_IO "${UWVM_UNITREE_PUBLIC_INCLUDE}/wasm/language/c++/include")
endif()
if(NOT IS_ABSOLUTE "${UWVM_UNITREE_FAST_IO}" OR
   NOT EXISTS "${UWVM_UNITREE_FAST_IO}/fast_io.h")
  message(FATAL_ERROR "Actual canonical C++ foundation/fast_io root required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_UNITREE_PUBLIC_INCLUDE}" OR
   NOT EXISTS "${UWVM_UNITREE_PUBLIC_INCLUDE}/platform/ros/c/roswasm/codec.h")
  message(FATAL_ERROR "Actual canonical SDK include root required for native peer")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(unitree_api REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(unitree-native-topic-loopback
  "${CMAKE_CURRENT_LIST_DIR}/native_topic_loopback.cpp")
target_compile_features(unitree-native-topic-loopback PRIVATE cxx_std_20)
target_include_directories(unitree-native-topic-loopback PRIVATE
  "${UWVM_UNITREE_API_GENERATED}" "${UWVM_UNITREE_PUBLIC_INCLUDE}"
  "${UWVM_UNITREE_PUBLIC_INCLUDE}/platform/ros/c"
  "${UWVM_UNITREE_FAST_IO}")
ament_target_dependencies(unitree-native-topic-loopback rclcpp unitree_api)
