cmake_minimum_required(VERSION 3.16)
project(uwvm_official_standard_native_contracts LANGUAGES CXX)
# This is an explicit native source build entry, never an automatic ROS test or
# exporter invocation. Root must own configuration/build/run and original PIDs,
# full SDK/source, ELF/maps, full MD, link inputs and every output/report.
set(UWVM_SOURCE "" CACHE PATH "Actual current complete uwvm source root")
set(UWVM_CUSTOM_NATIVE_ROS_VERSION "" CACHE STRING "Exactly 1 or 2")
set(UWVM_STANDARD_GENERATED "" CACHE PATH "Actual native-bound generated bundle")
option(UWVM_STANDARD_IDENTITY_ONLY "Build only the real native identity exporter" ON)
set(BUILD_TESTING OFF CACHE BOOL "Only explicitly selected owned contracts" FORCE)
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/src/uwvm2/imported/ros/ipc/custom_type.h")
  message(FATAL_ERROR "Select the actual frozen current UWVM_SOURCE")
endif()
add_executable(official-standard-native-identities native_identity.cc)
set(standard_targets official-standard-native-identities)
if(NOT UWVM_STANDARD_IDENTITY_ONLY)
  if(NOT IS_ABSOLUTE "${UWVM_STANDARD_GENERATED}" OR
     NOT EXISTS "${UWVM_STANDARD_GENERATED}/generation.json" OR
     NOT EXISTS "${UWVM_STANDARD_GENERATED}/manifest.hpp" OR
     NOT EXISTS "${UWVM_STANDARD_GENERATED}/native_ros${UWVM_CUSTOM_NATIVE_ROS_VERSION}.hpp" OR
     "${UWVM_STANDARD_GENERATED}" MATCHES "[;\"\\\r\n]")
    message(FATAL_ERROR "Select the actual native-export-bound generated bundle")
  endif()
  add_executable(official-standard-native-roundtrip native_roundtrip.cc)
  add_executable(official-standard-registry-contract registry_contract.cc)
  list(APPEND standard_targets official-standard-native-roundtrip official-standard-registry-contract)
endif()
foreach(target IN LISTS standard_targets)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=${UWVM_CUSTOM_NATIVE_ROS_VERSION})
  target_include_directories(${target} PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/src")
  if(target STREQUAL "official-standard-native-identities")
    set(native_file native_identity.cc)
  elseif(target STREQUAL "official-standard-native-roundtrip")
    set(native_file native_roundtrip.cc)
  else()
    set(native_file registry_contract.cc)
  endif()
  # These are requested controls. Only actual verbose compile argv and the
  # final effective MD/MF/MT/primary/header inputs qualify their consumption.
  target_compile_options(${target} PRIVATE -O1 -g0 -UNDEBUG -std=gnu++17 -MD
    -MF "${CMAKE_BINARY_DIR}/CMakeFiles/${target}.dir/${native_file}.o.d"
    -MT "${CMAKE_BINARY_DIR}/CMakeFiles/${target}.dir/${native_file}.o")
  target_link_options(${target} PRIVATE -Wl,-t)
  if(NOT target STREQUAL "official-standard-native-identities")
    target_include_directories(${target} PRIVATE "${UWVM_STANDARD_GENERATED}")
    target_compile_definitions(${target} PRIVATE UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_STANDARD_GENERATED}/manifest.hpp")
  endif()
endforeach()
if(UWVM_CUSTOM_NATIVE_ROS_VERSION STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs)
  foreach(target IN LISTS standard_targets)
    target_include_directories(${target} PRIVATE ${catkin_INCLUDE_DIRS})
    target_link_libraries(${target} PRIVATE ${catkin_LIBRARIES})
  endforeach()
elseif(UWVM_CUSTOM_NATIVE_ROS_VERSION STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(builtin_interfaces REQUIRED)
  find_package(rosidl_typesupport_cpp REQUIRED)
  foreach(target IN LISTS standard_targets)
    ament_target_dependencies(${target} PUBLIC std_msgs sensor_msgs builtin_interfaces rosidl_typesupport_cpp)
  endforeach()
else()
  message(FATAL_ERROR "Select exactly one actual native ROS major")
endif()
