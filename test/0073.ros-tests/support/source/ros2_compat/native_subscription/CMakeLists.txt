cmake_minimum_required(VERSION 3.16)
project(uwvm_native_rmw_subscription_contracts LANGUAGES CXX)
option(UWVM_NATIVE_SUBSCRIPTION_QUEUE_CONTRACT "Build SDK-free data queue contract" OFF)
option(UWVM_NATIVE_SUBSCRIPTION_REAL_SDK "Build real original-SDK witness" OFF)
get_filename_component(UWVM_NATIVE_SUBSCRIPTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23|26)$")
  message(FATAL_ERROR "Select an admitted actual C++ standard explicitly")
endif()
find_package(Threads REQUIRED)
function(uwvm_active_contract target)
  target_include_directories(${target} PRIVATE "${UWVM_NATIVE_SUBSCRIPTION_ROOT}/src")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
endfunction()
if(UWVM_NATIVE_SUBSCRIPTION_QUEUE_CONTRACT)
  add_executable(native-subscription-queue-contract queue_contract.cc)
  uwvm_active_contract(native-subscription-queue-contract)
endif()
if(UWVM_NATIVE_SUBSCRIPTION_REAL_SDK)
  find_package(ament_cmake REQUIRED)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    find_package(${dependency} REQUIRED)
  endforeach()
  if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
    message(FATAL_ERROR "Supply a real selected-SDK compile/link-probed D4 feature directory")
  endif()
  add_executable(native-subscription-sdk-contract native_subscription_contract.cc)
  uwvm_active_contract(native-subscription-sdk-contract)
  target_include_directories(native-subscription-sdk-contract PRIVATE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}")
  target_link_libraries(native-subscription-sdk-contract PRIVATE OpenSSL::Crypto)
  foreach(dependency IN ITEMS rclcpp rmw std_msgs geometry_msgs sensor_msgs rosidl_typesupport_introspection_cpp)
    ament_target_dependencies(native-subscription-sdk-contract ${dependency})
  endforeach()
endif()
