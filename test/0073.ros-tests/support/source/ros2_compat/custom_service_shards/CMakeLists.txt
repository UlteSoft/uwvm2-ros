cmake_minimum_required(VERSION 3.20)
project(uwvm_legacy_custom_service_shard_contract LANGUAGES CXX)
# Future explicitly owned real SDK build. No add_test or auto-run is installed.
set(UWVM_SOURCE_ROOT "" CACHE PATH "Exact composed reviewed source root")
set(UWVM_ROS2_GATEWAY_FEATURE_DIR "" CACHE PATH "Actual original SDK compile/link feature output")
set(UWVM_ROS_CUSTOM_REGISTRY_HEADER "" CACHE FILEPATH "Actual generated complete common manifest")
set(UWVM_ROS_CUSTOM_ROS2_HEADER "" CACHE FILEPATH "Actual generated complete native provider")
set(UWVM_ROS_CUSTOM_ROS2_PACKAGES "" CACHE STRING "Complete original interface package selection")
set(UWVM_LEGACY_SERVICE_OMIT_SHARD "" CACHE STRING "NEGATIVE LINK ONLY: exact one shard to omit")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE_ROOT}" OR NOT EXISTS "${UWVM_SOURCE_ROOT}/src/ros/gateway/ros2/custom_services.cmake")
  message(FATAL_ERROR "Supply the exact composed reviewed source root")
endif()
if(NOT IS_ABSOLUTE "${UWVM_ROS2_GATEWAY_FEATURE_DIR}" OR NOT EXISTS "${UWVM_ROS2_GATEWAY_FEATURE_DIR}/uwvm_ros2_features.h")
  message(FATAL_ERROR "Supply actual selected-SDK compile/link feature output")
endif()
if(NOT CMAKE_CXX_STANDARD MATCHES "^(17|20|23)$")
  message(FATAL_ERROR "Select the actual admitted original SDK C++ standard17/20/23")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include/fast_io_freestanding.h")
  message(FATAL_ERROR "The actual vendored fast_io include closure is required")
endif()
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(base_packages rclcpp rcl rmw rosidl_typesupport_cpp rosidl_typesupport_introspection_cpp std_msgs geometry_msgs sensor_msgs std_srvs rcl_interfaces)
foreach(package IN LISTS base_packages)
  find_package(${package} REQUIRED)
endforeach()
set(selected FALSE)
if(UWVM_ROS_CUSTOM_REGISTRY_HEADER OR UWVM_ROS_CUSTOM_ROS2_HEADER OR UWVM_ROS_CUSTOM_ROS2_PACKAGES)
  if(NOT UWVM_ROS_CUSTOM_REGISTRY_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_HEADER OR NOT UWVM_ROS_CUSTOM_ROS2_PACKAGES)
    message(FATAL_ERROR "Require the whole actual paired generated headers and explicit package list")
  endif()
  foreach(header IN ITEMS UWVM_ROS_CUSTOM_REGISTRY_HEADER UWVM_ROS_CUSTOM_ROS2_HEADER)
    if(NOT IS_ABSOLUTE "${${header}}" OR NOT EXISTS "${${header}}" OR IS_DIRECTORY "${${header}}")
      message(FATAL_ERROR "${header} must name an actual absolute generated header")
    endif()
    if("${${header}}" MATCHES "[;\"\\\\\r\n]")
      message(FATAL_ERROR "Unsupported quoted include-path character")
    endif()
  endforeach()
  set(seen "")
  foreach(package IN LISTS UWVM_ROS_CUSTOM_ROS2_PACKAGES)
    if(NOT package MATCHES "^[a-z][a-z0-9_]*$")
      message(FATAL_ERROR "Require explicit actual interface package identifiers")
    endif()
    list(FIND seen "${package}" duplicate)
    if(NOT duplicate EQUAL -1)
      message(FATAL_ERROR "Duplicate selected interface package")
    endif()
    list(APPEND seen "${package}")
    find_package(${package} REQUIRED)
  endforeach()
  set(selected TRUE)
endif()
add_executable(legacy-service-shard-contract source_contract.cc)
target_include_directories(legacy-service-shard-contract PRIVATE "${UWVM_SOURCE_ROOT}" "${UWVM_SOURCE_ROOT}/src"
  "${UWVM_SOURCE_ROOT}/sdk/origin/include" "${UWVM_SOURCE_ROOT}/src/ros/gateway/ros2" "${UWVM_ROS2_GATEWAY_FEATURE_DIR}"
  "${UWVM_SOURCE_ROOT}/third-parties/fast_io/include")
target_compile_options(legacy-service-shard-contract PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
# Keep the original ament plain link signature; do not mix keyword signatures.
ament_target_dependencies(legacy-service-shard-contract ${base_packages})
target_link_libraries(legacy-service-shard-contract Threads::Threads OpenSSL::Crypto)
if(selected)
  target_compile_definitions(legacy-service-shard-contract PRIVATE
    UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_ROS_CUSTOM_REGISTRY_HEADER}"
    UWVM_ROS_CUSTOM_ROS2_HEADER="${UWVM_ROS_CUSTOM_ROS2_HEADER}")
  ament_target_dependencies(legacy-service-shard-contract ${UWVM_ROS_CUSTOM_ROS2_PACKAGES})
  include("${UWVM_SOURCE_ROOT}/src/ros/gateway/ros2/custom_services.cmake")
  uwvm_ros2_custom_service_sources(legacy-service-shard-contract)
endif()
if(NOT UWVM_LEGACY_SERVICE_OMIT_SHARD STREQUAL "")
  if(NOT selected OR NOT UWVM_LEGACY_SERVICE_OMIT_SHARD MATCHES "^(0|[1-9][0-9]*)$" OR
     UWVM_LEGACY_SERVICE_OMIT_SHARD GREATER_EQUAL UWVM_ROS2_CUSTOM_SERVICE_SHARDS)
    message(FATAL_ERROR "Negative omission requires a complete custom selection and one actual shard ordinal")
  endif()
  get_target_property(complete_sources legacy-service-shard-contract SOURCES)
  set(omitted "${CMAKE_CURRENT_BINARY_DIR}/custom-service-shards/legacy_service_shard_contract/custom_services_shard_${UWVM_LEGACY_SERVICE_OMIT_SHARD}.cc")
  list(FIND complete_sources "${omitted}" present)
  if(present EQUAL -1)
    message(FATAL_ERROR "Expected generated shard source not found")
  endif()
  list(REMOVE_ITEM complete_sources "${omitted}")
  set_property(TARGET legacy-service-shard-contract PROPERTY SOURCES "${complete_sources}")
  message(STATUS "NONQUALIFYING NEGATIVE target deliberately omits ${omitted}")
endif()
