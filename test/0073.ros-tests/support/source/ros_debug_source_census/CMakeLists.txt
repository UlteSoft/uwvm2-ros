cmake_minimum_required(VERSION 3.20)
project(uwvm_ros_debug_source_contract LANGUAGES C CXX)
option(UWVM_ROS_DEBUG_CENSUS_CONTRACT_SOURCE "Compile actual Native contract OBJECT only" OFF)
option(UWVM_ROS_DEBUG_CENSUS_GUEST_SOURCE "Compile actual C11 Guest/DWARF fixture" OFF)
set(UWVM_SOURCE "" CACHE PATH "Exact reviewed complete Source root")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/src/uwvm2/uwvm/runtime/storage/builtin_ros_loader.h")
 message(FATAL_ERROR "Actual reviewed full Source tree including this port is required")
endif()
if(UWVM_ROS_DEBUG_CENSUS_CONTRACT_SOURCE)
 set(UWVM_ACTUAL_RUNTIME_USAGE_TARGET "" CACHE STRING "Existing real Runtime configuration/feature/include target")
 if(NOT TARGET "${UWVM_ACTUAL_RUNTIME_USAGE_TARGET}")
  message(FATAL_ERROR "Pass an existing genuine project Runtime usage target; no fake feature/ROS shims")
 endif()
 add_library(ros_debug_census_contract OBJECT identity_contract.cc)
 target_compile_features(ros_debug_census_contract PRIVATE cxx_std_23)
 target_include_directories(ros_debug_census_contract PRIVATE "${UWVM_SOURCE}/src")
 target_link_libraries(ros_debug_census_contract PRIVATE "${UWVM_ACTUAL_RUNTIME_USAGE_TARGET}")
endif()
if(UWVM_ROS_DEBUG_CENSUS_GUEST_SOURCE)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  message(FATAL_ERROR "Guest requires an actual WASI compiler/sysroot, separate from the Native build")
 endif()
 add_executable(ros_debug_census_guest guest.c)
 target_compile_features(ros_debug_census_guest PRIVATE c_std_11)
 target_include_directories(ros_debug_census_guest PRIVATE "${UWVM_SOURCE}/sdk/origin/include")
 target_compile_options(ros_debug_census_guest PRIVATE -g -O0)
 target_link_options(ros_debug_census_guest PRIVATE -Wl,--export-memory)
endif()
