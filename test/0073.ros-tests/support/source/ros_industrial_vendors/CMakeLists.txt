cmake_minimum_required(VERSION 3.20)
project(uwvm_industrial_contracts LANGUAGES CXX)
set(UWVM_SOURCE "" CACHE PATH "Explicit exact source root; this target does not run")
if(NOT IS_ABSOLUTE "${UWVM_SOURCE}" OR NOT EXISTS "${UWVM_SOURCE}/sdk/origin/include/roswasm/roswasmcpp.hpp")
  message(FATAL_ERROR "An actual current SDK source root is required")
endif()
add_executable(industrial_trajectory_contract trajectory_contract.cc)
target_compile_features(industrial_trajectory_contract PRIVATE cxx_std_20)
target_include_directories(industrial_trajectory_contract PRIVATE "${UWVM_SOURCE}" "${UWVM_SOURCE}/sdk/origin/include")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(industrial_trajectory_contract PRIVATE -UNDEBUG)
elseif(MSVC)
  target_compile_options(industrial_trajectory_contract PRIVATE /UNDEBUG)
endif()
