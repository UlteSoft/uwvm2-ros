cmake_minimum_required(VERSION 3.10)
project(uwvm_ros1_gateway LANGUAGES CXX)
set(UWVM_ROS1_CXX_STANDARD "17" CACHE STRING "Target C++ standard, at least the adapter's C++17 contract: 17, 20 or 23")
set_property(CACHE UWVM_ROS1_CXX_STANDARD PROPERTY STRINGS 17 20 23)
if(NOT UWVM_ROS1_CXX_STANDARD MATCHES "^(17|20|23)$")
    message(FATAL_ERROR "UWVM_ROS1_CXX_STANDARD must be 17, 20 or 23")
endif()
if((UWVM_ROS1_CXX_STANDARD STREQUAL "20" AND CMAKE_VERSION VERSION_LESS "3.12") OR
   (UWVM_ROS1_CXX_STANDARD STREQUAL "23" AND CMAKE_VERSION VERSION_LESS "3.20"))
    message(FATAL_ERROR "Selected C++ standard needs a newer host CMake: C++20 >=3.12, C++23 >=3.20")
endif()
set(CMAKE_CXX_STANDARD ${UWVM_ROS1_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The distro label is metadata, never a capability/backend selector. Prefer the
# selected catkin SDK; EXPLICIT also accepts a maintained pre-catkin SDK/overlay
# whose public headers and actual libraries pass the same compile/link probes.
# Neither mode runs a target program, which keeps cross sysroots usable.
set(UWVM_ROS1_DISCOVERY "CATKIN" CACHE STRING "ROS1 development SDK discovery: CATKIN or EXPLICIT")
set_property(CACHE UWVM_ROS1_DISCOVERY PROPERTY STRINGS CATKIN EXPLICIT)
set(UWVM_ROS1_INCLUDE_DIRS "" CACHE STRING "EXPLICIT mode: absolute selected SDK include directories")
set(UWVM_ROS1_LIBRARIES "" CACHE STRING "EXPLICIT mode: absolute selected SDK native libraries")
option(UWVM_ROS1_BUILD_PROBES "Build all historical native acceptance probes in addition to the Gateway" ON)

# Custom Topics, Services and Actions are an explicit trusted build selection.
# An ordinary build has
# an empty custom registry and creates no additional communication entities.
# Generate and review both headers from the same official package definitions,
# then select every required native catkin package explicitly. These options do
# not grant a ROS operation to a Guest; all usual CLI capability fields remain
# required. The historical TOPIC_MANIFEST macro also carries the reviewed
# Service and Action descriptors; its name is retained for build compatibility.
set(UWVM_ROS_CUSTOM_REGISTRY_HEADER "" CACHE FILEPATH "Generated common bounded ROS interface registry header")
set(UWVM_ROS_CUSTOM_ROS1_HEADER "" CACHE FILEPATH "Generated native ROS1 Topic, Service and Action provider header")
set(UWVM_ROS_CUSTOM_ROS1_PACKAGES "" CACHE STRING "Explicit semicolon-separated custom catkin interface packages")
if(UWVM_ROS_CUSTOM_REGISTRY_HEADER OR UWVM_ROS_CUSTOM_ROS1_HEADER OR UWVM_ROS_CUSTOM_ROS1_PACKAGES)
    if(NOT UWVM_ROS_CUSTOM_REGISTRY_HEADER OR NOT UWVM_ROS_CUSTOM_ROS1_HEADER OR NOT UWVM_ROS_CUSTOM_ROS1_PACKAGES)
        message(FATAL_ERROR "Custom ROS1 interfaces require both generated headers and an explicit package list")
    endif()
    foreach(_uwvm_custom_header UWVM_ROS_CUSTOM_REGISTRY_HEADER UWVM_ROS_CUSTOM_ROS1_HEADER)
        if(NOT EXISTS "${${_uwvm_custom_header}}" OR IS_DIRECTORY "${${_uwvm_custom_header}}")
            message(FATAL_ERROR "Missing custom interface header: ${_uwvm_custom_header}")
        endif()
        get_filename_component(${_uwvm_custom_header} "${${_uwvm_custom_header}}" REALPATH)
        # A preprocessor include string cannot represent these characters
        # safely. Fail during configuration instead of changing its meaning.
        if("${${_uwvm_custom_header}}" MATCHES "[\";\\\\\r\n]")
            message(FATAL_ERROR "Custom header path contains an unsupported preprocessor character")
        endif()
    endforeach()
    set(_uwvm_custom_packages_seen "")
    foreach(_uwvm_custom_package IN LISTS UWVM_ROS_CUSTOM_ROS1_PACKAGES)
        if(NOT "${_uwvm_custom_package}" MATCHES "^[a-z][a-z0-9_]*$")
            message(FATAL_ERROR "Invalid explicit custom catkin package name")
        endif()
        list(FIND _uwvm_custom_packages_seen "${_uwvm_custom_package}" _uwvm_custom_duplicate)
        if(NOT _uwvm_custom_duplicate EQUAL -1)
            message(FATAL_ERROR "Duplicate explicit custom catkin package name")
        endif()
        list(APPEND _uwvm_custom_packages_seen "${_uwvm_custom_package}")
    endforeach()
endif()
if(UWVM_ROS1_DISCOVERY STREQUAL "CATKIN")
    find_package(catkin REQUIRED COMPONENTS roscpp std_msgs geometry_msgs sensor_msgs std_srvs tf2 tf2_ros tf2_msgs actionlib actionlib_msgs actionlib_tutorials ${UWVM_ROS_CUSTOM_ROS1_PACKAGES})
elseif(UWVM_ROS1_DISCOVERY STREQUAL "EXPLICIT")
    if(NOT UWVM_ROS1_INCLUDE_DIRS OR NOT UWVM_ROS1_LIBRARIES)
        message(FATAL_ERROR "EXPLICIT SDK discovery requires both absolute include and native-library lists")
    endif()
    foreach(_uwvm_include IN LISTS UWVM_ROS1_INCLUDE_DIRS)
        if(NOT IS_ABSOLUTE "${_uwvm_include}" OR NOT IS_DIRECTORY "${_uwvm_include}")
            message(FATAL_ERROR "An explicit ROS1 include directory must be existing and absolute")
        endif()
    endforeach()
    foreach(_uwvm_library IN LISTS UWVM_ROS1_LIBRARIES)
        if(NOT IS_ABSOLUTE "${_uwvm_library}" OR NOT EXISTS "${_uwvm_library}" OR IS_DIRECTORY "${_uwvm_library}")
            message(FATAL_ERROR "An explicit ROS1 native library must be an existing absolute file")
        endif()
    endforeach()
    set(catkin_INCLUDE_DIRS ${UWVM_ROS1_INCLUDE_DIRS})
    set(catkin_LIBRARIES ${UWVM_ROS1_LIBRARIES})
else()
    message(FATAL_ERROR "UWVM_ROS1_DISCOVERY must be CATKIN or EXPLICIT")
endif()
add_library(uwvm-ros1-sdk INTERFACE)
target_include_directories(uwvm-ros1-sdk INTERFACE ../../src ${catkin_INCLUDE_DIRS})
target_link_libraries(uwvm-ros1-sdk INTERFACE ${catkin_LIBRARIES} pthread)
include(cmake/Compatibility.cmake)
add_executable(uwvm-ros1-gateway main.cc)
if(UWVM_ROS_CUSTOM_REGISTRY_HEADER)
    target_compile_definitions(uwvm-ros1-gateway PRIVATE
        UWVM_ROS_CUSTOM_TOPIC_MANIFEST="${UWVM_ROS_CUSTOM_REGISTRY_HEADER}"
        UWVM_ROS_CUSTOM_ROS1_HEADER="${UWVM_ROS_CUSTOM_ROS1_HEADER}")
    if(catkin_EXPORTED_TARGETS)
        add_dependencies(uwvm-ros1-gateway ${catkin_EXPORTED_TARGETS})
    endif()
endif()
target_compile_options(uwvm-ros1-gateway PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(uwvm-ros1-gateway PRIVATE uwvm-ros1-sdk)
if(UWVM_ROS1_BUILD_PROBES)
