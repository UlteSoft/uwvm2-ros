include(CheckCXXSourceCompiles)
function(_complete_api output)
 set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
 set(CMAKE_REQUIRED_INCLUDES "${UWVM_RMW_INCLUDE_DIR}")
 set(CMAKE_REQUIRED_LIBRARIES ${_native_targets} rosidl_runtime_c::rosidl_runtime_c)
 set(_flags "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_LINK_FLAGS} ${CMAKE_EXE_LINKER_FLAGS} ${CMAKE_REQUIRED_FLAGS} ${CMAKE_REQUIRED_LIBRARIES} ${CMAKE_REQUIRED_LINK_OPTIONS}")
 foreach(_config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL ${CMAKE_BUILD_TYPE} ${CMAKE_TRY_COMPILE_CONFIGURATION})
  string(TOUPPER "${_config}" _upper)
  string(APPEND _flags " ${CMAKE_CXX_FLAGS_${_upper}} ${CMAKE_EXE_LINKER_FLAGS_${_upper}}")
 endforeach()
 if(_flags MATCHES "--(allow-undefined|import-undefined|unresolved-symbols)")
  message(FATAL_ERROR "Permissive unresolved-symbol flags cannot prove a native link")
 endif()
 file(READ "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/probes/complete_api.cpp" _source)
 unset(_UWVM_DEFAULT_CONTROL_GATEWAY_COMPLETE_API)
 unset(_UWVM_DEFAULT_CONTROL_GATEWAY_COMPLETE_API CACHE)
 check_cxx_source_compiles("${_source}" _UWVM_DEFAULT_CONTROL_GATEWAY_COMPLETE_API)
 if(_UWVM_DEFAULT_CONTROL_GATEWAY_COMPLETE_API)
  set(${output} 1 PARENT_SCOPE)
 else()
  set(${output} 0 PARENT_SCOPE)
 endif()
 unset(_UWVM_DEFAULT_CONTROL_GATEWAY_COMPLETE_API CACHE)
endfunction()
